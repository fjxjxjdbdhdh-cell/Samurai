class SimpleNeuralNet:
    """Lightweight feed-forward neural network with online learning."""

    def __init__(self, input_size: int, hidden_sizes: List[int], output_size: int, lr: float = 0.001):
        self.lr = lr
        self.layers = []
        sizes = [input_size] + hidden_sizes + [output_size]
        for i in range(len(sizes) - 1):
            w = np.random.randn(sizes[i], sizes[i + 1]) * np.sqrt(2.0 / sizes[i])
            b = np.zeros((1, sizes[i + 1]))
            self.layers.append({"w": w, "b": b})

    def _relu(self, x):
        return np.maximum(0, x)

    def _relu_grad(self, x):
        return (x > 0).astype(float)

    def _sigmoid(self, x):
        return 1.0 / (1.0 + np.exp(-np.clip(x, -500, 500)))

    def _tanh(self, x):
        return np.tanh(x)

    def forward(self, x: np.ndarray) -> np.ndarray:
        self._activations = [x]
        current = x
        for i, layer in enumerate(self.layers):
            z = current @ layer["w"] + layer["b"]
            if i < len(self.layers) - 1:
                current = self._relu(z)
            else:
                current = self._tanh(z)
            self._activations.append(current)
        return current

    def predict(self, x: np.ndarray) -> np.ndarray:
        return self.forward(x)

    def train_step(self, x: np.ndarray, target: np.ndarray) -> float:
        out = self.forward(x)
        loss = np.mean((out - target) ** 2)
        delta = 2 * (out - target) / out.size
        delta = delta * (1 - out ** 2)
        for i in reversed(range(len(self.layers))):
            act = self._activations[i]
            dw = act.T @ delta
            db = np.sum(delta, axis=0, keepdims=True)
            self.layers[i]["w"] -= self.lr * dw
            self.layers[i]["b"] -= self.lr * db
            if i > 0:
                delta = delta @ self.layers[i]["w"].T
                delta = delta * self._relu_grad(self._activations[i])
        return float(loss)

    def get_weights(self) -> List[dict]:
        return [{"w": l["w"].tolist(), "b": l["b"].tolist()} for l in self.layers]

    def set_weights(self, weights: List[dict]):
        for i, w_dict in enumerate(weights):
            self.layers[i]["w"] = np.array(w_dict["w"])
            self.layers[i]["b"] = np.array(w_dict["b"])

class QLearningAgent:
    """Q-learning agent for trading decisions."""

    def __init__(self, state_size: int, action_size: int = 3):
        self.state_size = state_size
        self.action_size = action_size
        self.gamma = 0.95
        self.epsilon = 0.3
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
        self.lr = 0.001
        self.replay_buffer = deque(maxlen=10000)
        self.model = SimpleNeuralNet(state_size, [128, 64, 32], action_size, lr=self.lr)
        self.target_model = SimpleNeuralNet(state_size, [128, 64, 32], action_size, lr=self.lr)
        self.update_target_counter = 0
        self.update_target_freq = 100
        self.batch_size = 32
        self.total_steps = 0

    def get_action(self, state: np.ndarray, explore: bool = True) -> int:
        if explore and random.random() < self.epsilon:
            return random.randint(0, self.action_size - 1)
        q_values = self.model.predict(state.reshape(1, -1))
        return int(np.argmax(q_values[0]))

    def store_experience(self, state, action, reward, next_state, done):
        self.replay_buffer.append((
            state.copy(),
            action,
            reward,
            next_state.copy(),
            done,
        ))

    def train(self) -> float:
        if len(self.replay_buffer) < self.batch_size:
            return 0.0
        batch = random.sample(self.replay_buffer, self.batch_size)
        total_loss = 0.0
        for state, action, reward, next_state, done in batch:
            s = state.reshape(1, -1)
            ns = next_state.reshape(1, -1)
            current_q = self.model.predict(s)[0].copy()
            if done:
                target_q = reward
            else:
                next_q = self.target_model.predict(ns)[0]
                target_q = reward + self.gamma * np.max(next_q)
            current_q[action] = target_q
            loss = self.model.train_step(s, current_q.reshape(1, -1))
            total_loss += loss
        self.update_target_counter += 1
        if self.update_target_counter >= self.update_target_freq:
            self.target_model.set_weights(self.model.get_weights())
            self.update_target_counter = 0
        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay
        self.total_steps += 1
        return total_loss / self.batch_size

    def get_state_quality(self, state: np.ndarray) -> float:
        q_values = self.model.predict(state.reshape(1, -1))[0]
        return float(np.max(q_values))

class MarketIntelligence:
    """Analyzes market data and generates intelligence."""

    def __init__(self):
        self.symbol_data: Dict[str, dict] = {}
        self.regime_history: deque = deque(maxlen=100)
        self.volatility_window = 20
        self.trend_window = 50

    def update(self, symbol: str, tick: dict, candles: dict):
        if symbol not in self.symbol_data:
            self.symbol_data[symbol] = {
                "ticks": deque(maxlen=1000),
                "volatility": 0.0,
                "trend_strength": 0.0,
                "regime": "UNKNOWN",
                "support_levels": [],
                "resistance_levels": [],
                "momentum": 0.0,
                "volume_profile": 0.0,
                "last_analysis": None,
            }
        sd = self.symbol_data[symbol]
        sd["ticks"].append(tick)
        if candles.get("m15") is not None and len(candles["m15"]) > self.volatility_window:
            sd["volatility"] = self._calc_volatility(candles["m15"])
            sd["trend_strength"] = self._calc_trend_strength(candles["m15"])
            sd["momentum"] = self._calc_momentum(candles["m15"])
            sd["regime"] = self._detect_regime(candles)
            sd["support_levels"] = self._find_support_levels(candles["m15"])
            sd["resistance_levels"] = self._find_resistance_levels(candles["m15"])
            sd["volume_profile"] = self._calc_volume_profile(candles["m15"])
            sd["last_analysis"] = datetime.utcnow().isoformat()

    def _calc_volatility(self, candles) -> float:
        try:
            closes = np.array([c[4] for c in candles[-self.volatility_window:]])
            returns = np.diff(np.log(closes + 1e-10))
            return float(np.std(returns) * np.sqrt(252 * 24 * 4) * 100)
        except Exception:
            return 0.0

    def _calc_trend_strength(self, candles) -> float:
        try:
            closes = np.array([c[4] for c in candles[-self.trend_window:]])
            x = np.arange(len(closes))
            if len(closes) < 2:
                return 0.0
            slope, _ = np.polyfit(x, closes, 1)
            price_range = np.max(closes) - np.min(closes)
            if price_range == 0:
                return 0.0
            normalized = (slope * len(closes)) / price_range
            return float(np.clip(normalized, -1.0, 1.0))
        except Exception:
            return 0.0

    def _calc_momentum(self, candles, period: int = 14) -> float:
        try:
            closes = np.array([c[4] for c in candles[-period - 1:]])
            if len(closes) < period + 1:
                return 0.0
            return float((closes[-1] - closes[0]) / closes[0] * 100)
        except Exception:
            return 0.0

    def _detect_regime(self, candles: dict) -> str:
        try:
            m15 = candles.get("m15")
            if m15 is None or len(m15) < 50:
                return "RANGING"

            closes_m15 = np.array([c[4] for c in m15[-50:]])
            vol = self._calc_volatility(m15)
            trend = self._calc_trend_strength(m15)
            momentum = self._calc_momentum(m15)

            if vol > 30:
                return "VOLATILE"
            if trend > 0.5 and momentum > 0.1:
                return "TRENDING_UP"
            if trend < -0.5 and momentum < -0.1:
                return "TRENDING_DOWN"

            sma20 = np.mean(closes_m15[-20:])
            price = closes_m15[-1]
            dist_from_sma = abs(price - sma20) / sma20 * 100

            if dist_from_sma < 0.05:
                recent_high = np.max(closes_m15[-10:])
                recent_low = np.min(closes_m15[-10:])
                if (recent_high - recent_low) / recent_low * 100 > 0.3:
                    return "BREAKOUT"

            if abs(trend) < 0.2:
                return "RANGING"
            return "TRENDING_UP" if trend > 0 else "TRENDING_DOWN"
        except Exception:
            return "RANGING"

    def _find_support_levels(self, candles) -> List[float]:
        try:
            lows = np.array([c[3] for c in candles[-100:]])
            supports = []
            for i in range(2, len(lows) - 2):
                if lows[i] < lows[i-1] and lows[i] < lows[i+1]:
                    if lows[i] < lows[i-2] and lows[i] < lows[i+2]:
                        supports.append(float(lows[i]))
            return sorted(supports)[-5:] if supports else []
        except Exception:
            return []

    def _find_resistance_levels(self, candles) -> List[float]:
        try:
            highs = np.array([c[2] for c in candles[-100:]])
            resistances = []
            for i in range(2, len(highs) - 2):
                if highs[i] > highs[i-1] and highs[i] > highs[i+1]:
                    if highs[i] > highs[i-2] and highs[i] > highs[i+2]:
                        resistances.append(float(highs[i]))
            return sorted(resistances)[:5] if resistances else []
        except Exception:
            return []

    def _calc_volume_profile(self, candles) -> float:
        try:
            volumes = np.array([c[5] for c in candles[-20:]])
            if len(volumes) < 2:
                return 0.0
            avg_vol = np.mean(volumes[:-1])
            if avg_vol == 0:
                return 0.0
            return float(volumes[-1] / avg_vol)
        except Exception:
            return 0.0

    def get_technical_indicators(self, symbol: str, candles: dict) -> dict:
        indicators = {}
        try:
            for tf_name, tf_candles in candles.items():
                if tf_candles is None or len(tf_candles) < 20:
                    continue
                closes = np.array([c[4] for c in tf_candles])
                highs = np.array([c[2] for c in tf_candles])
                lows = np.array([c[3] for c in tf_candles])
                volumes = np.array([c[5] for c in tf_candles])
                indicators[tf_name] = {
                    "sma_20": float(np.mean(closes[-20:])),
                    "sma_50": float(np.mean(closes[-50:])) if len(closes) >= 50 else float(np.mean(closes)),
                    "sma_200": float(np.mean(closes[-200:])) if len(closes) >= 200 else float(np.mean(closes)),
                    "ema_12": float(self._ema(closes, 12)),
                    "ema_26": float(self._ema(closes, 26)),
                    "rsi": float(self._rsi(closes)),
                    "macd": float(self._ema(closes, 12) - self._ema(closes, 26)),
                    "bollinger_upper": float(np.mean(closes[-20:]) + 2 * np.std(closes[-20:])),
                    "bollinger_lower": float(np.mean(closes[-20:]) - 2 * np.std(closes[-20:])),
                    "atr": float(self._atr(highs, lows, closes)),
                    "volume_avg": float(np.mean(volumes[-20:])),
                    "current_close": float(closes[-1]),
                    "current_high": float(highs[-1]),
                    "current_low": float(lows[-1]),
                }
        except Exception as e:
            logger.error(f"Technical indicators error: {e}")
        return indicators

    def _ema(self, data: np.ndarray, period: int) -> float:
        if len(data) < period:
            return float(np.mean(data))
        k = 2.0 / (period + 1)
        ema = float(data[0])
        for v in data[1:]:
            ema = float(v) * k + ema * (1 - k)
        return ema

    def _rsi(self, closes: np.ndarray, period: int = 14) -> float:
        if len(closes) < period + 1:
            return 50.0
        deltas = np.diff(closes[-period - 1:])
        gains = np.where(deltas > 0, deltas, 0)
        losses = np.where(deltas < 0, -deltas, 0)
        avg_gain = np.mean(gains[-period:])
        avg_loss = np.mean(losses[-period:])
        if avg_loss == 0:
            return 100.0
        rs = avg_gain / avg_loss
        return float(100 - (100 / (1 + rs)))

    def _atr(self, highs: np.ndarray, lows: np.ndarray, closes: np.ndarray, period: int = 14) -> float:
        if len(closes) < 2:
            return 0.0
        trs = []
        for i in range(1, min(period + 1, len(closes))):
            tr = max(
                highs[-i] - lows[-i],
                abs(highs[-i] - closes[-i - 1]),
                abs(lows[-i] - closes[-i - 1]),
            )
            trs.append(tr)
        return float(np.mean(trs)) if trs else 0.0

    def build_state_vector(self, symbol: str, candles: dict, account: dict) -> np.ndarray:
        features = []
        try:
            indicators = self.get_technical_indicators(symbol, candles)
            sd = self.symbol_data.get(symbol, {})
            for tf in ["m1", "m15", "h1", "h4"]:
                ind = indicators.get(tf, {})
                close = ind.get("current_close", 0)
                sma20 = ind.get("sma_20", close)
                sma50 = ind.get("sma_50", close)
                rsi = ind.get("rsi", 50)
                macd = ind.get("macd", 0)
                atr = ind.get("atr", 0)
                bb_upper = ind.get("bollinger_upper", close)
                bb_lower = ind.get("bollinger_lower", close)
                bb_width = (bb_upper - bb_lower) / (close + 1e-10)
                bb_pos = (close - bb_lower) / (bb_upper - bb_lower + 1e-10)
                features.extend([
                    (close - sma20) / (sma20 + 1e-10),
                    (sma20 - sma50) / (sma50 + 1e-10),
                    (rsi - 50) / 50,
                    macd / (close + 1e-10) * 1000,
                    atr / (close + 1e-10) * 1000,
                    bb_width,
                    bb_pos - 0.5,
                ])
            features.append(sd.get("volatility", 0) / 100)
            features.append(sd.get("trend_strength", 0))
            features.append(sd.get("momentum", 0) / 10)
            features.append(sd.get("volume_profile", 1) - 1)

            regime_encoding = [0.0] * len(REGIMES)
            regime = sd.get("regime", "RANGING")
            if regime in REGIMES:
                regime_encoding[REGIMES.index(regime)] = 1.0
            features.extend(regime_encoding)

            balance = account.get("balance", 10000)
            equity = account.get("equity", 10000)
            free_margin = account.get("free_margin", 10000)
            features.append((equity - balance) / (balance + 1e-10))
            features.append(free_margin / (balance + 1e-10))

            state = np.array(features, dtype=np.float32)
            state = np.nan_to_num(state, nan=0.0, posinf=1.0, neginf=-1.0)
            state = np.clip(state, -5.0, 5.0)
            return state
        except Exception as e:
            logger.error(f"Build state vector error: {e}")
            return np.zeros(50, dtype=np.float32)

    def get_market_summary(self, symbol: str) -> dict:
        sd = self.symbol_data.get(symbol, {})
        return {
            "symbol": symbol,
            "regime": sd.get("regime", "UNKNOWN"),
            "volatility": sd.get("volatility", 0.0),
            "trend_strength": sd.get("trend_strength", 0.0),
            "momentum": sd.get("momentum", 0.0),
            "support_levels": sd.get("support_levels", []),
            "resistance_levels": sd.get("resistance_levels", []),
            "volume_profile": sd.get("volume_profile", 0.0),
            "last_analysis": sd.get("last_analysis"),
        }

class RiskEngine:
    def __init__(self):
        self.risk_per_trade: float = 0.01
        self.max_positions: int = 5
        self.max_daily_loss: float = 0.05
        self.max_drawdown: float = 0.15
        self.trailing_stop_multiplier: float = 2.0
        self.atr_sl_multiplier: float = 1.5
        self.atr_tp_multiplier: float = 3.0
        self.daily_loss_tracker: float = 0.0
        self.daily_loss_reset: datetime = datetime.utcnow()
        self.peak_equity: float = 0.0
        self.current_drawdown: float = 0.0
        self.trade_history_risk: deque = deque(maxlen=50)
        self.consecutive_losses: int = 0
        self.consecutive_wins: int = 0
        self.win_rate: float = 0.5
        self.avg_win: float = 0.0
        self.avg_loss: float = 0.0
        self.risk_multiplier: float = 1.0

    def reset_daily_if_needed(self):
        now = datetime.utcnow()
        if now.date() > self.daily_loss_reset.date():
            self.daily_loss_tracker = 0.0
            self.daily_loss_reset = now

    def update_equity(self, equity: float, balance: float):
        self.reset_daily_if_needed()
        if equity > self.peak_equity:
            self.peak_equity = equity
        if self.peak_equity > 0:
            self.current_drawdown = (self.peak_equity - equity) / self.peak_equity

    def record_trade_result(self, profit: float, balance: float):
        self.reset_daily_if_needed()
        self.daily_loss_tracker += profit
        self.trade_history_risk.append({
            "profit": profit,
            "timestamp": datetime.utcnow().isoformat(),
        })
        if profit < 0:
            self.consecutive_losses += 1
            self.consecutive_wins = 0
        else:
            self.consecutive_wins += 1
            self.consecutive_losses = 0

        profits = [t["profit"] for t in self.trade_history_risk]
        wins = [p for p in profits if p > 0]
        losses = [p for p in profits if p < 0]
        total = len(profits)
        self.win_rate = len(wins) / total if total > 0 else 0.5
        self.avg_win = float(np.mean(wins)) if wins else 0.0
        self.avg_loss = float(np.mean(losses)) if losses else 0.0
        self._update_risk_multiplier()

    def _update_risk_multiplier(self):
        base = 1.0
        if self.current_drawdown > 0.10:
            base *= 0.5
        elif self.current_drawdown > 0.07:
            base *= 0.7
        elif self.current_drawdown > 0.05:
            base *= 0.85
        if self.consecutive_losses >= 3:
            base *= max(0.3, 1.0 - (self.consecutive_losses - 2) * 0.15)
        if self.consecutive_wins >= 5 and self.win_rate > 0.6:
            base *= min(1.5, 1.0 + (self.consecutive_wins - 4) * 0.05)
        self.risk_multiplier = float(np.clip(base, 0.2, 2.0))

    def can_open_trade(self, account: dict, positions: list, symbol: str) -> Tuple[bool, str]:
        self.reset_daily_if_needed()
        balance = account.get("balance", 0)
        equity = account.get("equity", 0)
        free_margin = account.get("free_margin", 0)

        if balance <= 0:
            return False, "Zero balance"

        self.update_equity(equity, balance)

        if self.current_drawdown >= self.max_drawdown:
            return False, f"Max drawdown reached: {self.current_drawdown:.2%}"

        daily_loss_pct = abs(self.daily_loss_tracker) / balance if self.daily_loss_tracker < 0 else 0
        if daily_loss_pct >= self.max_daily_loss:
            return False, f"Daily loss limit reached: {daily_loss_pct:.2%}"

        if len(positions) >= self.max_positions:
            return False, f"Max positions ({self.max_positions}) reached"

        open_symbols = [p["symbol"] for p in positions]
        symbol_count = open_symbols.count(symbol)
        if symbol_count >= 2:
            return False, f"Already have {symbol_count} positions on {symbol}"

        if free_margin < balance * 0.1:
            return False, "Insufficient free margin"

        return True, "OK"

    def calculate_position_size(self, symbol: str, account: dict,
                                 entry_price: float, sl_price: float,
                                 atr: float = 0.0) -> float:
        try:
            balance = account.get("balance", 10000)
            risk_amount = balance * self.risk_per_trade * self.risk_multiplier
            if sl_price <= 0 or entry_price <= 0:
                risk_per_pip = atr if atr > 0 else entry_price * 0.001
            else:
                risk_per_pip = abs(entry_price - sl_price)
            if risk_per_pip == 0:
                risk_per_pip = entry_price * 0.001

            pip_value = self._get_pip_value(symbol, entry_price)
            if pip_value <= 0:
                pip_value = 10.0

            volume = risk_amount / (risk_per_pip * pip_value / self._get_pip_size(symbol))
            volume = round(float(np.clip(volume, 0.01, 10.0)), 2)
            return volume
        except Exception as e:
            logger.error(f"Position size calc error: {e}")
            return 0.01

    def _get_pip_size(self, symbol: str) -> float:
        sym = symbol.upper()
        if "JPY" in sym:
            return 0.01
        if any(x in sym for x in ["XAU", "GOLD"]):
            return 0.1
        if any(x in sym for x in ["BTC", "ETH"]):
            return 1.0
        return 0.0001

    def _get_pip_value(self, symbol: str, price: float) -> float:
        sym = symbol.upper()
        if "JPY" in sym:
            return 9.09
        if any(x in sym for x in ["XAU", "GOLD"]):
            return 1.0
        if any(x in sym for x in ["BTC"]):
            return 0.1
        return 10.0

    def calculate_dynamic_sl_tp(self, symbol: str, order_type: str,
                                  entry_price: float, atr: float,
                                  regime: str, confidence: float,
                                  support_levels: List[float],
                                  resistance_levels: List[float]) -> Tuple[float, float]:
        try:
            if atr <= 0:
                atr = entry_price * 0.001

            sl_mult = self.atr_sl_multiplier
            tp_mult = self.atr_tp_multiplier

            regime_adjustments = {
                "TRENDING_UP":   (1.2, 3.5),
                "TRENDING_DOWN": (1.2, 3.5),
                "RANGING":       (1.0, 2.0),
                "VOLATILE":      (2.0, 3.0),
                "BREAKOUT":      (1.5, 4.0),
                "REVERSAL":      (1.0, 2.5),
            }
            sl_mult, tp_mult = regime_adjustments.get(regime, (sl_mult, tp_mult))

            tp_mult *= (0.5 + confidence)
            sl_mult *= (1.0 + (1.0 - confidence) * 0.3)

            sl_distance = atr * sl_mult
            tp_distance = atr * tp_mult

            if order_type == "BUY":
                sl = entry_price - sl_distance
                tp = entry_price + tp_distance
                if support_levels:
                    nearest_support = max([s for s in support_levels if s < entry_price], default=sl)
                    if abs(nearest_support - sl) < sl_distance * 0.5:
                        sl = nearest_support - atr * 0.3
                if resistance_levels:
                    nearest_resistance = min([r for r in resistance_levels if r > entry_price], default=tp)
                    if nearest_resistance < tp:
                        tp = nearest_resistance - atr * 0.2
            else:
                sl = entry_price + sl_distance
                tp = entry_price - tp_distance
                if resistance_levels:
                    nearest_resistance = min([r for r in resistance_levels if r > entry_price], default=sl)
                    if abs(nearest_resistance - sl) < sl_distance * 0.5:
                        sl = nearest_resistance + atr * 0.3
                if support_levels:
                    nearest_support = max([s for s in support_levels if s < entry_price], default=tp)
                    if nearest_support > tp:
                        tp = nearest_support + atr * 0.2

            min_sl_dist = entry_price * 0.0005
            if abs(sl - entry_price) < min_sl_dist:
                sl = entry_price - min_sl_dist if order_type == "BUY" else entry_price + min_sl_dist
            if abs(tp - entry_price) < min_sl_dist * 2:
                tp = entry_price + min_sl_dist * 2 if order_type == "BUY" else entry_price - min_sl_dist * 2

            return round(float(sl), 5), round(float(tp), 5)
        except Exception as e:
            logger.error(f"SL/TP calculation error: {e}")
            fallback = entry_price * 0.002
            if order_type == "BUY":
                return round(entry_price - fallback, 5), round(entry_price + fallback * 2, 5)
            return round(entry_price + fallback, 5), round(entry_price - fallback * 2, 5)

    def should_trail_stop(self, position: dict, current_price: float, atr: float) -> Optional[float]:
        try:
            order_type = position.get("order_type", "BUY")
            open_price = position.get("open_price", current_price)
            current_sl = position.get("sl", 0.0)
            if atr <= 0:
                atr = current_price * 0.001

            trail_distance = atr * self.trailing_stop_multiplier

            if order_type == "BUY":
                profit_pips = current_price - open_price
                if profit_pips < atr * 0.5:
                    return None
                new_sl = current_price - trail_distance
                if new_sl > current_sl + atr * 0.1:
                    return round(new_sl, 5)
            else:
                profit_pips = open_price - current_price
                if profit_pips < atr * 0.5:
                    return None
                new_sl = current_price + trail_distance
                if new_sl < current_sl - atr * 0.1 or current_sl == 0:
                    return round(new_sl, 5)
            return None
        except Exception as e:
            logger.error(f"Trail stop error: {e}")
            return None

    def get_risk_report(self) -> dict:
        return {
            "risk_per_trade": self.risk_per_trade,
            "risk_multiplier": self.risk_multiplier,
            "max_positions": self.max_positions,
            "current_drawdown": round(self.current_drawdown, 4),
            "daily_loss": round(self.daily_loss_tracker, 2),
            "consecutive_losses": self.consecutive_losses,
            "consecutive_wins": self.consecutive_wins,
            "win_rate": round(self.win_rate, 3),
            "avg_win": round(self.avg_win, 2),
            "avg_loss": round(self.avg_loss, 2),
            "expectancy": round(
                self.win_rate * self.avg_win + (1 - self.win_rate) * self.avg_loss, 2
            ),
        }

class StrategyEngine:
    def __init__(self):
        self.strategy_performance: Dict[str, dict] = {
            s: {"wins": 0, "losses": 0, "profit": 0.0, "count": 0}
            for s in STRATEGIES
        }
        self.current_strategy: str = "TREND_FOLLOW"
        self.strategy_weights: Dict[str, float] = {s: 1.0 for s in STRATEGIES}
        self.mode_history: deque = deque(maxlen=20)

    def select_strategy(self, regime: str, volatility: float,
                        trend_strength: float, momentum: float,
                        rsi: float, win_rate: float) -> str:
        scores: Dict[str, float] = {}

        for strategy in STRATEGIES:
            base = self.strategy_weights.get(strategy, 1.0)
            perf = self.strategy_performance[strategy]
            if perf["count"] > 5:
                wr = perf["wins"] / perf["count"]
                perf_bonus = (wr - 0.5) * 2.0
            else:
                perf_bonus = 0.0

            if strategy == "TREND_FOLLOW":
                regime_score = 1.5 if regime in ["TRENDING_UP", "TRENDING_DOWN"] else 0.3
                scores[strategy] = base + regime_score + abs(trend_strength) + perf_bonus
            elif strategy == "MEAN_REVERT":
                regime_score = 1.5 if regime == "RANGING" else 0.2
                overbought = 1.0 if rsi > 70 or rsi < 30 else 0.3
                scores[strategy] = base + regime_score + overbought + perf_bonus
            elif strategy == "BREAKOUT":
                regime_score = 1.5 if regime == "BREAKOUT" else 0.3
                scores[strategy] = base + regime_score + volatility / 20 + perf_bonus
            elif strategy == "SCALP":
                regime_score = 1.2 if regime == "RANGING" else 0.5
                scores[strategy] = base + regime_score + (1.0 - volatility / 30) + perf_bonus
            elif strategy == "MOMENTUM":
                regime_score = 1.3 if regime in ["TRENDING_UP", "TRENDING_DOWN", "BREAKOUT"] else 0.2
                scores[strategy] = base + regime_score + abs(momentum) / 5 + perf_bonus
            elif strategy == "SWING":
                regime_score = 1.0 if regime in ["TRENDING_UP", "TRENDING_DOWN", "REVERSAL"] else 0.5
                scores[strategy] = base + regime_score + abs(trend_strength) * 0.5 + perf_bonus
            elif strategy == "DEFENSIVE":
                scores[strategy] = base + (1.0 if volatility > 25 else 0.1) + perf_bonus

        best = max(scores, key=lambda k: scores[k])
        self.current_strategy = best
        return best

    def select_trading_mode(self, regime: str, volatility: float) -> str:
        if volatility > 30:
            mode = "SHORT_TERM"
        elif volatility > 20:
            if regime in ["TRENDING_UP", "TRENDING_DOWN"]:
                mode = "SWING"
            else:
                mode = "SHORT_TERM"
        elif volatility > 10:
            mode = "SWING"
        else:
            mode = "LONG_TERM"

        if self.current_strategy == "SCALP":
            mode = "HFT"
        elif self.current_strategy == "SWING":
            mode = "SWING"

        self.mode_history.append(mode)
        return mode

    def get_cycle_delay_for_mode(self, mode: str) -> float:
        delays = {
            "HFT": 2.0,
            "SHORT_TERM": 15.0,
            "SWING": 60.0,
            "LONG_TERM": 300.0,
        }
        return delays.get(mode, 15.0)

    def record_result(self, strategy: str, profit: float):
        if strategy not in self.strategy_performance:
            self.strategy_performance[strategy] = {"wins": 0, "losses": 0, "profit": 0.0, "count": 0}
        perf = self.strategy_performance[strategy]
        perf["count"] += 1
        perf["profit"] += profit
        if profit > 0:
            perf["wins"] += 1
        else:
            perf["losses"] += 1
        if perf["count"] >= 5:
            wr = perf["wins"] / perf["count"]
            profit_factor = min(2.0, max(0.3, 1.0 + perf["profit"] / (abs(perf["profit"]) + 1)))
            self.strategy_weights[strategy] = wr * profit_factor

    def generate_signal(self, strategy: str, indicators: dict,
                         regime: str, order_type_hint: str = None) -> dict:
        signal = {"action": "HOLD", "confidence": 0.0, "reasons": []}

        try:
            m15 = indicators.get("m15", {})
            h1 = indicators.get("h1", {})
            if not m15 and not h1:
                return signal

            close = m15.get("current_close", 0) or h1.get("current_close", 0)
            if close <= 0:
                return signal

            rsi_m15 = m15.get("rsi", 50)
            rsi_h1 = h1.get("rsi", 50)
            macd_m15 = m15.get("macd", 0)
            macd_h1 = h1.get("macd", 0)
            sma20 = m15.get("sma_20", close)
            sma50 = m15.get("sma_50", close)
            sma200 = h1.get("sma_200", close)
            ema12 = m15.get("ema_12", close)
            ema26 = m15.get("ema_26", close)
            bb_upper = m15.get("bollinger_upper", close)
            bb_lower = m15.get("bollinger_lower", close)

            bull_score = 0.0
            bear_score = 0.0
            reasons = []

            if strategy == "TREND_FOLLOW":
                if close > sma20 > sma50:
                    bull_score += 1.5
                    reasons.append("Price above SMA20>SMA50")
                if close < sma20 < sma50:
                    bear_score += 1.5
                    reasons.append("Price below SMA20<SMA50")
                if close > sma200:
                    bull_score += 1.0
                    reasons.append("Price above SMA200")
                else:
                    bear_score += 1.0
                    reasons.append("Price below SMA200")
                if macd_m15 > 0 and macd_h1 > 0:
                    bull_score += 1.0
                    reasons.append("MACD bullish M15+H1")
                elif macd_m15 < 0 and macd_h1 < 0:
                    bear_score += 1.0
                    reasons.append("MACD bearish M15+H1")
                if 40 < rsi_m15 < 65:
                    bull_score += 0.5
                elif 35 < rsi_m15 < 60:
                    bear_score += 0.3

            elif strategy == "MEAN_REVERT":
                if rsi_m15 < 30 and rsi_h1 < 40:
                    bull_score += 2.0
                    reasons.append("RSI oversold M15+H1")
                if rsi_m15 > 70 and rsi_h1 > 60:
                    bear_score += 2.0
                    reasons.append("RSI overbought M15+H1")
                if close <= bb_lower:
                    bull_score += 1.5
                    reasons.append("Price at Bollinger lower")
                if close >= bb_upper:
                    bear_score += 1.5
                    reasons.append("Price at Bollinger upper")
                if ema12 > ema26 and rsi_m15 < 45:
                    bull_score += 0.5
                if ema12 < ema26 and rsi_m15 > 55:
                    bear_score += 0.5

            elif strategy == "BREAKOUT":
                bb_width = (bb_upper - bb_lower) / (close + 1e-10)
                if bb_width < 0.005:
                    if close > bb_upper and macd_m15 > 0:
                        bull_score += 2.5
                        reasons.append("Breakout above Bollinger + MACD")
                    elif close < bb_lower and macd_m15 < 0:
                        bear_score += 2.5
                        reasons.append("Breakout below Bollinger + MACD")
                if close > sma20 and ema12 > ema26:
                    bull_score += 1.0
                elif close < sma20 and ema12 < ema26:
                    bear_score += 1.0

            elif strategy == "SCALP":
                if rsi_m15 < 35 and close > sma20 * 0.9995:
                    bull_score += 1.5
                    reasons.append("Scalp buy: RSI dip + near SMA")
                if rsi_m15 > 65 and close < sma20 * 1.0005:
                    bear_score += 1.5
                    reasons.append("Scalp sell: RSI peak + near SMA")
                if macd_m15 > 0 and ema12 > ema26:
                    bull_score += 0.8
                if macd_m15 < 0 and ema12 < ema26:
                    bear_score += 0.8

            elif strategy == "MOMENTUM":
                momentum_m15 = (close - sma50) / (sma50 + 1e-10) * 100
                if momentum_m15 > 0.2 and macd_m15 > 0 and rsi_m15 > 50:
                    bull_score += 2.0
                    reasons.append(f"Momentum bullish {momentum_m15:.2f}%")
                if momentum_m15 < -0.2 and macd_m15 < 0 and rsi_m15 < 50:
                    bear_score += 2.0
                    reasons.append(f"Momentum bearish {momentum_m15:.2f}%")
                if close > sma20 and ema12 > ema26:
                    bull_score += 0.8
                elif close < sma20 and ema12 < ema26:
                    bear_score += 0.8

            elif strategy == "SWING":
                if close > sma200 and sma20 > sma50 and rsi_h1 > 40 and rsi_h1 < 70:
                    bull_score += 2.0
                    reasons.append("Swing buy: aligned MAs + H1 RSI")
                if close < sma200 and sma20 < sma50 and rsi_h1 < 60 and rsi_h1 > 30:
                    bear_score += 2.0
                    reasons.append("Swing sell: aligned MAs + H1 RSI")
                h4_rsi = indicators.get("h4", {}).get("rsi", 50)
                if h4_rsi < 40 and bull_score > 0:
                    bull_score += 1.0
                    reasons.append("H4 RSI supports buy")
                if h4_rsi > 60 and bear_score > 0:
                    bear_score += 1.0
                    reasons.append("H4 RSI supports sell")

            elif strategy == "DEFENSIVE":
                if rsi_m15 < 25 and rsi_h1 < 30 and close < bb_lower:
                    bull_score += 2.5
                    reasons.append("DEFENSIVE: extreme oversold")
                if rsi_m15 > 75 and rsi_h1 > 70 and close > bb_upper:
                    bear_score += 2.5
                    reasons.append("DEFENSIVE: extreme overbought")

            if regime == "TRENDING_UP":
                bull_score *= 1.2
            elif regime == "TRENDING_DOWN":
                bear_score *= 1.2
            elif regime == "VOLATILE":
                bull_score *= 0.6
                bear_score *= 0.6

            total = bull_score + bear_score
            if total == 0:
                return signal

            if bull_score > bear_score and bull_score >= 2.0:
                confidence = min(0.95, bull_score / (total + 1e-10))
                signal = {"action": "BUY", "confidence": confidence, "reasons": reasons}
            elif bear_score > bull_score and bear_score >= 2.0:
                confidence = min(0.95, bear_score / (total + 1e-10))
                signal = {"action": "SELL", "confidence": confidence, "reasons": reasons}
            else:
                signal = {"action": "HOLD", "confidence": 0.0, "reasons": reasons}

        except Exception as e:
            logger.error(f"Generate signal error: {e}")

        return signal

class MemoryEngine:
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.short_term: deque = deque(maxlen=1000)
        self.episode_buffer: deque = deque(maxlen=500)
        self.pattern_memory: Dict[str, list] = {}
        self.market_memory: Dict[str, deque] = {}
        self.reward_history: deque = deque(maxlen=500)
        self.total_trades: int = 0
        self.total_profit: float = 0.0
        self.best_trades: list = []
        self.worst_trades: list = []

    def store_experience(self, state: np.ndarray, action: int,
                          reward: float, next_state: np.ndarray,
                          done: bool, metadata: dict = None):
        experience = {
            "state": state,
            "action": action,
            "reward": reward,
            "next_state": next_state,
            "done": done,
            "metadata": metadata or {},
            "timestamp": datetime.utcnow().isoformat(),
        }
        self.short_term.append(experience)
        self.episode_buffer.append(experience)
        self.reward_history.append(reward)

    def record_trade(self, trade: dict):
        self.total_trades += 1
        profit = trade.get("profit", 0)
        self.total_profit += profit

        if profit > 0:
            self.best_trades.append(trade)
            self.best_trades = sorted(self.best_trades, key=lambda x: x.get("profit", 0), reverse=True)[:10]
        else:
            self.worst_trades.append(trade)
            self.worst_trades = sorted(self.worst_trades, key=lambda x: x.get("profit", 0))[:10]

        symbol = trade.get("symbol", "UNKNOWN")
        strategy = trade.get("strategy", "UNKNOWN")
        pattern_key = f"{symbol}_{strategy}"
        if pattern_key not in self.pattern_memory:
            self.pattern_memory[pattern_key] = []
        self.pattern_memory[pattern_key].append({
            "profit": profit,
            "regime": trade.get("regime", "UNKNOWN"),
            "timestamp": datetime.utcnow().isoformat(),
        })
        if len(self.pattern_memory[pattern_key]) > 100:
            self.pattern_memory[pattern_key] = self.pattern_memory[pattern_key][-100:]

    def get_pattern_quality(self, symbol: str, strategy: str, regime: str) -> float:
        key = f"{symbol}_{strategy}"
        patterns = self.pattern_memory.get(key, [])
        if not patterns:
            return 0.5
        regime_patterns = [p for p in patterns if p.get("regime") == regime]
        if not regime_patterns:
            regime_patterns = patterns[-20:]
        profits = [p["profit"] for p in regime_patterns]
        if not profits:
            return 0.5
        win_rate = len([p for p in profits if p > 0]) / len(profits)
        return float(np.clip(win_rate, 0.0, 1.0))

    def calculate_reward(self, trade_result: dict) -> float:
        profit = trade_result.get("profit", 0)
        balance = trade_result.get("balance", 10000)
        duration_hours = trade_result.get("duration_hours", 1)

        if balance <= 0:
            return -1.0

        profit_pct = profit / balance

        if profit_pct > 0:
            reward = np.log(1 + profit_pct * 100)
        else:
            reward = -np.log(1 + abs(profit_pct) * 100) * 1.5

        if duration_hours > 48 and profit <= 0:
            reward -= 0.5

        if duration_hours > 0 and profit > 0:
            reward += min(0.3, profit_pct / duration_hours * 100)

        return float(np.clip(reward, -3.0, 3.0))

    def sample_batch(self, batch_size: int) -> list:
        if len(self.episode_buffer) < batch_size:
            return list(self.episode_buffer)
        return random.sample(list(self.episode_buffer), batch_size)

    def get_recent_performance(self, n: int = 20) -> dict:
        rewards = list(self.reward_history)[-n:]
        if not rewards:
            return {"avg_reward": 0.0, "trend": "NEUTRAL"}
        avg = float(np.mean(rewards))
        trend = "IMPROVING" if len(rewards) > 5 and np.mean(rewards[-5:]) > np.mean(rewards[:5]) else \
                "DECLINING" if len(rewards) > 5 and np.mean(rewards[-5:]) < np.mean(rewards[:5]) else "NEUTRAL"
        return {"avg_reward": avg, "trend": trend, "samples": len(rewards)}

    def get_summary(self) -> dict:
        recent_perf = self.get_recent_performance()
        return {
            "total_trades": self.total_trades,
            "total_profit": round(self.total_profit, 2),
            "buffer_size": len(self.episode_buffer),
            "patterns_learned": len(self.pattern_memory),
            "recent_performance": recent_perf,
            "best_trade": self.best_trades[0] if self.best_trades else None,
            "worst_trade": self.worst_trades[0] if self.worst_trades else None,
        }

# LEARNING ENGINE
# ─────────────────────────────────────────────
class LearningEngine:
    def __init__(self, state_size: int = 50):
        self.state_size = state_size
        self.agents: Dict[str, QLearningAgent] = {}
        self.global_agent = QLearningAgent(state_size, action_size=3)
        self.learning_active = True
        self.total_training_steps = 0
        self.loss_history: deque = deque(maxlen=1000)
        self.performance_tracker: Dict[str, deque] = {}
        self.error_log: deque = deque(maxlen=100)
        self.adaptation_rate = 0.01
        self.learning_cycles = 0

    def get_agent(self, symbol: str) -> QLearningAgent:
        if symbol not in self.agents:
            self.agents[symbol] = QLearningAgent(self.state_size, action_size=3)
            logger.info(f"Created new RL agent for {symbol}")
        return self.agents[symbol]

    async def train_agents(self, memory: MemoryEngine) -> dict:
        results = {}
        try:
            batch = memory.sample_batch(64)
            if len(batch) < 8:
                return results

            for exp in batch:
                state = exp["state"]
                action = exp["action"]
                reward = exp["reward"]
                next_state = exp["next_state"]
                done = exp["done"]
                symbol = exp.get("metadata", {}).get("symbol")

                self.global_agent.store_experience(state, action, reward, next_state, done)
                if symbol:
                    agent = self.get_agent(symbol)
                    agent.store_experience(state, action, reward, next_state, done)

            global_loss = self.global_agent.train()
            self.loss_history.append(global_loss)
            results["global_loss"] = global_loss

            symbol_losses = {}
            for symbol, agent in self.agents.items():
                loss = agent.train()
                symbol_losses[symbol] = loss

            results["symbol_losses"] = symbol_losses
            self.total_training_steps += 1
            self.learning_cycles += 1

            if self.learning_cycles % 50 == 0:
                self._adapt_learning_rates()

        except Exception as e:
            logger.error(f"Train agents error: {e}")
            self.error_log.append({"error": str(e), "timestamp": datetime.utcnow().isoformat()})
        return results

    def _adapt_learning_rates(self):
        recent_losses = list(self.loss_history)[-50:]
        if not recent_losses:
            return
        avg_loss = np.mean(recent_losses)
        if avg_loss > 0.1:
            for agent in self.agents.values():
                agent.model.lr = max(0.0001, agent.model.lr * 0.95)
            self.global_agent.model.lr = max(0.0001, self.global_agent.model.lr * 0.95)
        elif avg_loss < 0.001:
            for agent in self.agents.values():
                agent.model.lr = min(0.01, agent.model.lr * 1.05)

    def get_action_recommendation(self, symbol: str, state: np.ndarray) -> Tuple[int, float]:
        agent = self.get_agent(symbol)
        action = agent.get_action(state, explore=False)
        quality = agent.get_state_quality(state)
        global_action = self.global_agent.get_action(state, explore=False)
        global_quality = self.global_agent.get_state_quality(state)

        if len(agent.replay_buffer) > 100:
            final_action = action
            final_quality = (quality * 0.6 + global_quality * 0.4)
        else:
            final_action = global_action
            final_quality = global_quality

        return final_action, float(np.clip(final_quality, 0.0, 1.0))

    def get_training_stats(self) -> dict:
        losses = list(self.loss_history)
        return {
            "total_steps": self.total_training_steps,
            "learning_cycles": self.learning_cycles,
            "current_loss": float(np.mean(losses[-10:])) if losses else 0.0,
            "trend_loss": "IMPROVING" if len(losses) > 20 and np.mean(losses[-10:]) < np.mean(losses[-20:-10]) else "STABLE",
            "agents_count": len(self.agents),
            "global_epsilon": round(self.global_agent.epsilon, 4),
            "errors": len(self.error_log),
        }

# SELF DEVELOPMENT ENGINE
# ─────────────────────────────────────────────
class SelfDevelopmentEngine:
    def __init__(self):
        self.knowledge_base: dict = {
            "market_patterns": {},
            "news_sentiment": {},
            "economic_events": [],
            "strategy_insights": {},
            "last_update": None,
        }
        self.update_interval = 3600
        self.last_update: Optional[datetime] = None
        self.development_log: deque = deque(maxlen=200)
        self.sentiment_scores: Dict[str, float] = {}
        self.economic_calendar: list = []
        self.insight_buffer: deque = deque(maxlen=50)
        self.improvement_suggestions: list = []

    async def run_development_cycle(self, market_intelligence: MarketIntelligence,
                                     strategy_engine: StrategyEngine,
                                     memory_engine: MemoryEngine):
        try:
            now = datetime.utcnow()
            if self.last_update and (now - self.last_update).seconds < 300:
                return

            await self._analyze_own_performance(memory_engine, strategy_engine)
            await self._update_market_knowledge(market_intelligence)
            await self._fetch_external_signals()
            await self._generate_insights(memory_engine, strategy_engine)
            await self._improve_strategy_weights(strategy_engine, memory_engine)

            self.last_update = now
            self.knowledge_base["last_update"] = now.isoformat()
            self.development_log.append({
                "timestamp": now.isoformat(),
                "event": "Development cycle completed",
            })
            logger.info("Self-development cycle completed")
        except Exception as e:
            logger.error(f"Self development cycle error: {e}")

    async def _analyze_own_performance(self, memory: MemoryEngine, strategy: StrategyEngine):
        try:
            summary = memory.get_summary()
            recent_perf = summary.get("recent_performance", {})
            avg_reward = recent_perf.get("avg_reward", 0.0)
            trend = recent_perf.get("trend", "NEUTRAL")

            for strat_name, perf in strategy.strategy_performance.items():
                if perf["count"] >= 10:
                    wr = perf["wins"] / perf["count"]
                    profit = perf["profit"]
                    key = strat_name
                    if key not in self.knowledge_base["strategy_insights"]:
                        self.knowledge_base["strategy_insights"][key] = {}
                    self.knowledge_base["strategy_insights"][key] = {
                        "win_rate": wr,
                        "total_profit": profit,
                        "count": perf["count"],
                        "quality": wr * (1 + max(0, profit / 1000)),
                    }

            if trend == "DECLINING":
                self.improvement_suggestions.append({
                    "type": "RISK_REDUCTION",
                    "reason": "Performance declining",
                    "timestamp": datetime.utcnow().isoformat(),
                })
            elif trend == "IMPROVING" and avg_reward > 0.5:
                self.improvement_suggestions.append({
                    "type": "RISK_INCREASE",
                    "reason": "Strong performance detected",
                    "timestamp": datetime.utcnow().isoformat(),
                })

            if len(self.improvement_suggestions) > 20:
                self.improvement_suggestions = self.improvement_suggestions[-20:]
        except Exception as e:
            logger.error(f"Analyze own performance error: {e}")

    async def _generate_insights(self, memory: MemoryEngine, strategy: StrategyEngine):
        try:
            for symbol, patterns in memory.pattern_memory.items():
                if len(patterns) < 10:
                    continue
                recent = patterns[-20:]
                profits = [p["profit"] for p in recent]
                if not profits:
                    continue
                avg_profit = float(np.mean(profits))
                win_rate = len([p for p in profits if p > 0]) / len(profits)
                insight = {
                    "symbol_strategy": symbol,
                    "avg_profit": avg_profit,
                    "win_rate": win_rate,
                    "sample_size": len(recent),
                    "assessment": "POSITIVE" if avg_profit > 0 and win_rate > 0.5 else "NEGATIVE",
                    "timestamp": datetime.utcnow().isoformat(),
                }
                self.insight_buffer.append(insight)
        except Exception as e:
            logger.error(f"Generate insights error: {e}")

    async def _improve_strategy_weights(self, strategy_engine: StrategyEngine,
                                         memory_engine: MemoryEngine):
        try:
            insights_map = {}
            for insight in self.insight_buffer:
                key = insight["symbol_strategy"]
                if key not in insights_map:
                    insights_map[key] = []
                insights_map[key].append(insight)

            for strat_name in STRATEGIES:
                relevant_insights = [
                    ins for key, ins_list in insights_map.items()
                    for ins in ins_list if strat_name in key
                ]
                if not relevant_insights:
                    continue
                avg_wr = float(np.mean([i["win_rate"] for i in relevant_insights]))
                current_weight = strategy_engine.strategy_weights.get(strat_name, 1.0)
                target_weight = avg_wr * 2.0
                new_weight = current_weight * 0.9 + target_weight * 0.1
                strategy_engine.strategy_weights[strat_name] = float(np.clip(new_weight, 0.1, 3.0))
        except Exception as e:
            logger.error(f"Improve strategy weights error: {e}")

# DECISION ENGINE

class DecisionEngine:
    def __init__(self):
        self.min_confidence = 0.60
        self.signal_confirmation_required = 2
        self.last_decisions: Dict[str, dict] = {}
        self.decision_cooldown: Dict[str, datetime] = {}
        self.cooldown_seconds = 60
        self.decision_history: deque = deque(maxlen=500)
        self.total_decisions = 0
        self.executed_decisions = 0

    def is_in_cooldown(self, symbol: str) -> bool:
        last = self.decision_cooldown.get(symbol)
        if last is None:
            return False
        return (datetime.utcnow() - last).total_seconds() < self.cooldown_seconds

    def set_cooldown(self, symbol: str):
        self.decision_cooldown[symbol] = datetime.utcnow()

    def make_decision(
        self,
        symbol: str,
        state: np.ndarray,
        signal: dict,
        rl_action: int,
        rl_quality: float,
        market_data: dict,
        positions: list,
        account: dict,
        risk_engine: RiskEngine,
        strategy: str,
        regime: str,
        indicators: dict,
        pattern_quality: float,
        sentiment: float,
    ) -> Optional[dict]:
        self.total_decisions += 1
        try:
            signal_action = signal.get("action", "HOLD")
            signal_confidence = signal.get("signal_confidence", signal.get("confidence", 0.0))
            signal_reasons = signal.get("reasons", [])

            rl_action_map = {0: "HOLD", 1: "BUY", 2: "SELL"}
            rl_signal = rl_action_map.get(rl_action, "HOLD")

            if self.is_in_cooldown(symbol):
                return None

            votes = {"BUY": 0.0, "SELL": 0.0, "HOLD": 0.0}

            if signal_action in votes:
                votes[signal_action] += signal_confidence * 0.45

            if rl_signal in votes:
                votes[rl_signal] += rl_quality * 0.30

            if pattern_quality > 0.55:
                if signal_action == "BUY":
                    votes["BUY"] += pattern_quality * 0.15
                elif signal_action == "SELL":
                    votes["SELL"] += pattern_quality * 0.15

            if sentiment > 0.2:
                votes["BUY"] += abs(sentiment) * 0.10
            elif sentiment < -0.2:
                votes["SELL"] += abs(sentiment) * 0.10

            if regime == "VOLATILE":
                votes["BUY"] *= 0.5
                votes["SELL"] *= 0.5
                votes["HOLD"] += 0.3
            elif regime == "TRENDING_UP":
                votes["BUY"] *= 1.2
                votes["SELL"] *= 0.7
            elif regime == "TRENDING_DOWN":
                votes["SELL"] *= 1.2
                votes["BUY"] *= 0.7

            best_action = max(votes, key=lambda k: votes[k])
            final_confidence = votes[best_action]

            if best_action == "HOLD" or final_confidence < self.min_confidence:
                return None

            can_open, reason = risk_engine.can_open_trade(account, positions, symbol)
            if not can_open:
                self.decision_history.append({
                    "symbol": symbol,
                    "action": "BLOCKED",
                    "reason": reason,
                    "timestamp": datetime.utcnow().isoformat(),
                })
                return None

            m15 = indicators.get("m15", {})
            current_price = m15.get("current_close", 0)
            if current_price <= 0:
                return None

            atr = m15.get("atr", current_price * 0.001)
            if atr <= 0:
                atr = current_price * 0.001

            market_summary = {}
            if symbol in market_data:
                md = market_data[symbol]
                if isinstance(md, dict):
                    market_summary = md

            support_levels = []
            resistance_levels = []

            sl, tp = risk_engine.calculate_dynamic_sl_tp(
                symbol=symbol,
                order_type=best_action,
                entry_price=current_price,
                atr=atr,
                regime=regime,
                confidence=final_confidence,
                support_levels=support_levels,
                resistance_levels=resistance_levels,
            )

            volume = risk_engine.calculate_position_size(
                symbol=symbol,
                account=account,
                entry_price=current_price,
                sl_price=sl,
                atr=atr,
            )

            if volume < 0.01:
                return None

            reasoning_parts = [
                f"Action: {best_action} on {symbol}",
                f"Confidence: {final_confidence:.2%}",
                f"Strategy: {strategy}",
                f"Regime: {regime}",
                f"TA Signal: {signal_action} ({signal_confidence:.2%})",
                f"RL Signal: {rl_signal} ({rl_quality:.2%})",
                f"Pattern Quality: {pattern_quality:.2%}",
                f"Sentiment: {sentiment:+.2f}",
            ]
            reasoning_parts.extend(signal_reasons)
            full_reasoning = " | ".join(reasoning_parts)

            decision = {
                "symbol": symbol,
                "action": f"OPEN_{best_action}",
                "execute": True,
                "volume": volume,
                "sl": sl,
                "tp": tp,
                "confidence": final_confidence,
                "strategy": strategy,
                "market_regime": regime,
                "reasoning": full_reasoning,
                "votes": votes,
                "atr": atr,
                "entry_price": current_price,
                "timestamp": datetime.utcnow().isoformat(),
            }

            self.set_cooldown(symbol)
            self.executed_decisions += 1
            self.last_decisions[symbol] = decision
            self.decision_history.append({
                "symbol": symbol,
                "action": f"OPEN_{best_action}",
                "confidence": final_confidence,
                "timestamp": datetime.utcnow().isoformat(),
            })

            logger.info(
                f"Decision: OPEN_{best_action} {symbol} "
                f"vol={volume} sl={sl} tp={tp} conf={final_confidence:.2%}"
            )
            return decision

        except Exception as e:
            logger.error(f"Decision engine error: {e}")
            return None

    def make_position_management_decision(
        self,
        position: dict,
        current_price: float,
        indicators: dict,
        regime: str,
        risk_engine: RiskEngine,
        account: dict,
    ) -> Optional[dict]:
        try:
            ticket = position.get("ticket")
            order_type = position.get("order_type", "BUY")
            open_price = position.get("open_price", current_price)
            current_sl = position.get("sl", 0.0)
            current_tp = position.get("tp", 0.0)
            profit = position.get("profit", 0.0)
            symbol = position.get("symbol", "")

            m15 = indicators.get("m15", {})
            atr = m15.get("atr", current_price * 0.001)
            rsi = m15.get("rsi", 50)
            macd = m15.get("macd", 0)

            if atr <= 0:
                atr = current_price * 0.001

            new_sl = risk_engine.should_trail_stop(position, current_price, atr)
            if new_sl is not None:
                return {
                    "action": "MODIFY_SLTP",
                    "execute": True,
                    "ticket": ticket,
                    "new_sl": new_sl,
                    "new_tp": current_tp,
                    "symbol": symbol,
                    "reasoning": f"Trailing stop moved to {new_sl:.5f} | ATR={atr:.5f}",
                    "confidence": 0.9,
                }

            if order_type == "BUY":
                adverse_move = open_price - current_price
            else:
                adverse_move = current_price - open_price

            if adverse_move > atr * 3.0:
                return {
                    "action": "CLOSE",
                    "execute": True,
                    "ticket": ticket,
                    "symbol": symbol,
                    "reasoning": f"Emergency close: adverse move {adverse_move:.5f} > 3x ATR {atr * 3:.5f}",
                    "confidence": 0.95,
                }

            if order_type == "BUY":
                if rsi > 78 and macd < 0 and regime in ["TRENDING_DOWN", "REVERSAL"]:
                    return {
                        "action": "CLOSE",
                        "execute": True,
                        "ticket": ticket,
                        "symbol": symbol,
                        "reasoning": f"Signal reversal: RSI={rsi:.1f} overbought + MACD={macd:.6f} bearish + regime={regime}",
                        "confidence": 0.80,
                    }
            else:
                if rsi < 22 and macd > 0 and regime in ["TRENDING_UP", "REVERSAL"]:
                    return {
                        "action": "CLOSE",
                        "execute": True,
                        "ticket": ticket,
                        "symbol": symbol,
                        "reasoning": f"Signal reversal: RSI={rsi:.1f} oversold + MACD={macd:.6f} bullish + regime={regime}",
                        "confidence": 0.80,
                    }

            if order_type == "BUY" and regime == "TRENDING_UP" and rsi > 55:
                new_tp_candidate = current_price + atr * 2.5
                if current_tp > 0 and new_tp_candidate > current_tp + atr * 0.5:
                    return {
                        "action": "MODIFY_SLTP",
                        "execute": True,
                        "ticket": ticket,
                        "new_sl": current_sl,
                        "new_tp": round(new_tp_candidate, 5),
                        "symbol": symbol,
                        "reasoning": f"TP extended to {new_tp_candidate:.5f} on strong uptrend",
                        "confidence": 0.75,
                    }
            elif order_type == "SELL" and regime == "TRENDING_DOWN" and rsi < 45:
                new_tp_candidate = current_price - atr * 2.5
                if current_tp > 0 and new_tp_candidate < current_tp - atr * 0.5:
                    return {
                        "action": "MODIFY_SLTP",
                        "execute": True,
                        "ticket": ticket,
                        "new_sl": current_sl,
                        "new_tp": round(new_tp_candidate, 5),
                        "symbol": symbol,
                        "reasoning": f"TP extended to {new_tp_candidate:.5f} on strong downtrend",
                        "confidence": 0.75,
                    }

            return None

        except Exception as e:
            logger.error(f"Position management decision error: {e}")
            return None

    def get_stats(self) -> dict:
        return {
            "total_decisions": self.total_decisions,
            "executed_decisions": self.executed_decisions,
            "execution_rate": (
                self.executed_decisions / self.total_decisions
                if self.total_decisions > 0 else 0.0
            ),
            "symbols_in_cooldown": [
                sym for sym, ts in self.decision_cooldown.items()
                if (datetime.utcnow() - ts).total_seconds() < self.cooldown_seconds
            ],
            "recent_decisions": list(self.decision_history)[-10:],
        }

# PERFORMANCE ANALYTICS ENGINE
# ─────────────────────────────────────────────
class PerformanceAnalyticsEngine:
    def __init__(self):
        self.equity_curve: deque = deque(maxlen=10000)
        self.daily_returns: deque = deque(maxlen=365)
        self.trade_metrics: deque = deque(maxlen=1000)
        self.symbol_performance: Dict[str, dict] = {}
        self.regime_performance: Dict[str, dict] = {}
        self.hourly_performance: Dict[int, dict] = {}
        self.weekly_performance: Dict[int, dict] = {}
        self.max_consecutive_wins: int = 0
        self.max_consecutive_losses: int = 0
        self._current_consecutive_wins: int = 0
        self._current_consecutive_losses: int = 0
        self.peak_equity: float = 0.0
        self.max_drawdown_ever: float = 0.0
        self.sharpe_ratio: float = 0.0
        self.sortino_ratio: float = 0.0
        self.profit_factor: float = 0.0
        self.calmar_ratio: float = 0.0
        self.last_analytics_update: Optional[datetime] = None

    def record_equity_point(self, equity: float, balance: float, timestamp: datetime):
        self.equity_curve.append({
            "equity": equity,
            "balance": balance,
            "timestamp": timestamp.isoformat(),
            "drawdown": (self.peak_equity - equity) / (self.peak_equity + 1e-10) if self.peak_equity > 0 else 0.0,
        })
        if equity > self.peak_equity:
            self.peak_equity = equity
        current_dd = (self.peak_equity - equity) / (self.peak_equity + 1e-10)
        if current_dd > self.max_drawdown_ever:
            self.max_drawdown_ever = current_dd

    def record_trade_result(self, trade: dict):
        try:
            profit = trade.get("profit", 0.0)
            symbol = trade.get("symbol", "UNKNOWN")
            strategy = trade.get("strategy", "UNKNOWN")
            regime = trade.get("regime", "UNKNOWN")
            open_time_str = trade.get("open_time", datetime.utcnow().isoformat())
            close_time_str = trade.get("close_time", datetime.utcnow().isoformat())

            try:
                open_dt = datetime.fromisoformat(open_time_str)
                close_dt = datetime.fromisoformat(close_time_str)
                duration_minutes = (close_dt - open_dt).total_seconds() / 60
            except Exception:
                duration_minutes = 60.0
                open_dt = datetime.utcnow()

            hour = open_dt.hour
            weekday = open_dt.weekday()

            metric = {
                "profit": profit,
                "symbol": symbol,
                "strategy": strategy,
                "regime": regime,
                "duration_minutes": duration_minutes,
                "hour": hour,
                "weekday": weekday,
                "timestamp": datetime.utcnow().isoformat(),
                "is_win": profit > 0,
            }
            self.trade_metrics.append(metric)

            if profit > 0:
                self._current_consecutive_wins += 1
                self._current_consecutive_losses = 0
                if self._current_consecutive_wins > self.max_consecutive_wins:
                    self.max_consecutive_wins = self._current_consecutive_wins
            else:
                self._current_consecutive_losses += 1
                self._current_consecutive_wins = 0
                if self._current_consecutive_losses > self.max_consecutive_losses:
                    self.max_consecutive_losses = self._current_consecutive_losses

            if symbol not in self.symbol_performance:
                self.symbol_performance[symbol] = {
                    "trades": 0, "wins": 0, "losses": 0,
                    "total_profit": 0.0, "avg_duration": 0.0,
                    "best_trade": 0.0, "worst_trade": 0.0,
                }
            sp = self.symbol_performance[symbol]
            sp["trades"] += 1
            sp["total_profit"] += profit
            if profit > 0:
                sp["wins"] += 1
                sp["best_trade"] = max(sp["best_trade"], profit)
            else:
                sp["losses"] += 1
                sp["worst_trade"] = min(sp["worst_trade"], profit)
            sp["avg_duration"] = (
                (sp["avg_duration"] * (sp["trades"] - 1) + duration_minutes) / sp["trades"]
            )

            if regime not in self.regime_performance:
                self.regime_performance[regime] = {
                    "trades": 0, "wins": 0, "total_profit": 0.0,
                }
            rp = self.regime_performance[regime]
            rp["trades"] += 1
            rp["total_profit"] += profit
            if profit > 0:
                rp["wins"] += 1

            if hour not in self.hourly_performance:
                self.hourly_performance[hour] = {
                    "trades": 0, "wins": 0, "total_profit": 0.0,
                }
            hp = self.hourly_performance[hour]
            hp["trades"] += 1
            hp["total_profit"] += profit
            if profit > 0:
                hp["wins"] += 1

            if weekday not in self.weekly_performance:
                self.weekly_performance[weekday] = {
                    "trades": 0, "wins": 0, "total_profit": 0.0,
                }
            wp = self.weekly_performance[weekday]
            wp["trades"] += 1
            wp["total_profit"] += profit
            if profit > 0:
                wp["wins"] += 1

            self._update_ratios()

        except Exception as e:
            logger.error(f"Record trade result analytics error: {e}")

    def _update_ratios(self):
        try:
            trades = list(self.trade_metrics)
            if len(trades) < 5:
                return

            profits = [t["profit"] for t in trades]
            wins = [p for p in profits if p > 0]
            losses = [p for p in profits if p < 0]

            gross_profit = sum(wins)
            gross_loss = abs(sum(losses))
            self.profit_factor = gross_profit / gross_loss if gross_loss > 0 else float(gross_profit > 0) * 99.0

            if len(profits) > 1:
                mean_r = float(np.mean(profits))
                std_r = float(np.std(profits))
                self.sharpe_ratio = (mean_r / (std_r + 1e-10)) * np.sqrt(252)

                downside = [p for p in profits if p < 0]
                if downside:
                    downside_std = float(np.std(downside))
                    self.sortino_ratio = (mean_r / (downside_std + 1e-10)) * np.sqrt(252)

            ann_return = float(np.sum(profits)) / max(1, len(profits)) * 252
            if self.max_drawdown_ever > 0:
                self.calmar_ratio = ann_return / self.max_drawdown_ever
            else:
                self.calmar_ratio = 0.0

            self.last_analytics_update = datetime.utcnow()

        except Exception as e:
            logger.error(f"Update ratios error: {e}")

    def get_best_trading_hours(self) -> List[int]:
        hour_scores = {}
        for hour, data in self.hourly_performance.items():
            if data["trades"] >= 3:
                wr = data["wins"] / data["trades"]
                score = wr * data["total_profit"]
                hour_scores[hour] = score
        if not hour_scores:
            return list(range(6, 22))
        sorted_hours = sorted(hour_scores, key=lambda h: hour_scores[h], reverse=True)
        return sorted_hours[:12]

    def get_best_regimes(self) -> List[str]:
        regime_scores = {}
        for regime, data in self.regime_performance.items():
            if data["trades"] >= 3:
                wr = data["wins"] / data["trades"]
                regime_scores[regime] = wr * (data["total_profit"] + 1)
        if not regime_scores:
            return REGIMES
        return sorted(regime_scores, key=lambda r: regime_scores[r], reverse=True)

    def get_full_report(self) -> dict:
        trades = list(self.trade_metrics)
        total = len(trades)
        if total == 0:
            return {"status": "no_data", "total_trades": 0}

        profits = [t["profit"] for t in trades]
        wins = [p for p in profits if p > 0]
        losses = [p for p in profits if p < 0]

        return {
            "total_trades": total,
            "win_rate": round(len(wins) / total, 4) if total > 0 else 0.0,
            "total_profit": round(sum(profits), 2),
            "avg_profit": round(float(np.mean(profits)), 2),
            "avg_win": round(float(np.mean(wins)), 2) if wins else 0.0,
            "avg_loss": round(float(np.mean(losses)), 2) if losses else 0.0,
            "best_trade": round(max(profits), 2) if profits else 0.0,
            "worst_trade": round(min(profits), 2) if profits else 0.0,
            "profit_factor": round(self.profit_factor, 3),
            "sharpe_ratio": round(self.sharpe_ratio, 3),
            "sortino_ratio": round(self.sortino_ratio, 3),
            "calmar_ratio": round(self.calmar_ratio, 3),
            "max_drawdown_ever": round(self.max_drawdown_ever, 4),
            "max_consecutive_wins": self.max_consecutive_wins,
            "max_consecutive_losses": self.max_consecutive_losses,
            "symbol_performance": {
                sym: {
                    "trades": d["trades"],
                    "win_rate": round(d["wins"] / d["trades"], 3) if d["trades"] > 0 else 0.0,
                    "total_profit": round(d["total_profit"], 2),
                    "avg_duration_min": round(d["avg_duration"], 1),
                    "best_trade": round(d["best_trade"], 2),
                    "worst_trade": round(d["worst_trade"], 2),
                }
                for sym, d in self.symbol_performance.items()
            },
            "regime_performance": {
                reg: {
                    "trades": d["trades"],
                    "win_rate": round(d["wins"] / d["trades"], 3) if d["trades"] > 0 else 0.0,
                    "total_profit": round(d["total_profit"], 2),
                }
                for reg, d in self.regime_performance.items()
            },
            "best_hours": self.get_best_trading_hours(),
            "best_regimes": self.get_best_regimes(),
            "equity_curve_points": len(self.equity_curve),
            "last_update": self.last_analytics_update.isoformat() if self.last_analytics_update else None,
        }

    def get_equity_curve(self, last_n: int = 200) -> list:
        curve = list(self.equity_curve)
        return curve[-last_n:] if len(curve) > last_n else curve

# SIGNAL AGGREGATOR

class SignalAggregator:
    def __init__(self):
        self.timeframe_weights = {
            "m1":  0.10,
            "m15": 0.30,
            "h1":  0.35,
            "h4":  0.25,
        }
        self.signal_history: Dict[str, deque] = {}
        self.divergence_log: deque = deque(maxlen=100)

    def aggregate(
        self,
        symbol: str,
        indicators: Dict[str, dict],
        regime: str,
        strategy: str,
        strategy_engine: "StrategyEngine",
    ) -> dict:
        if symbol not in self.signal_history:
            self.signal_history[symbol] = deque(maxlen=50)

        tf_signals = {}
        for tf, weight in self.timeframe_weights.items():
            tf_ind = {tf: indicators.get(tf, {})}
            if not tf_ind.get(tf):
                continue
            sig = strategy_engine.generate_signal(
                strategy=strategy,
                indicators=tf_ind,
                regime=regime,
            )
            tf_signals[tf] = {
                "action": sig.get("action", "HOLD"),
                "confidence": sig.get("confidence", 0.0),
                "weight": weight,
                "reasons": sig.get("reasons", []),
            }

        if not tf_signals:
            return {"action": "HOLD", "confidence": 0.0, "reasons": [], "tf_signals": {}}

        weighted_votes = {"BUY": 0.0, "SELL": 0.0, "HOLD": 0.0}
        all_reasons = []
        total_weight = 0.0

        for tf, sig_data in tf_signals.items():
            action = sig_data["action"]
            conf = sig_data["confidence"]
            weight = sig_data["weight"]
            if action in weighted_votes:
                weighted_votes[action] += conf * weight
            total_weight += weight
            all_reasons.extend(sig_data["reasons"])

        if total_weight > 0:
            for k in weighted_votes:
                weighted_votes[k] /= total_weight

        best_action = max(weighted_votes, key=lambda k: weighted_votes[k])
        best_confidence = weighted_votes[best_action]

        actions = [s["action"] for s in tf_signals.values()]
        agreement_rate = actions.count(best_action) / len(actions)

        if agreement_rate < 0.5:
            best_confidence *= 0.6
            all_reasons.append(f"Low TF agreement: {agreement_rate:.0%}")

        if agreement_rate >= 0.75 and best_action != "HOLD":
            best_confidence = min(0.95, best_confidence * 1.15)
            all_reasons.append(f"High TF agreement: {agreement_rate:.0%}")

        divergence = self._detect_divergence(tf_signals)
        if divergence:
            self.divergence_log.append({
                "symbol": symbol,
                "divergence": divergence,
                "timestamp": datetime.utcnow().isoformat(),
            })
            best_confidence *= 0.75

        aggregated = {
            "action": best_action if best_confidence >= 0.55 else "HOLD",
            "signal_confidence": round(best_confidence, 4),
            "confidence": round(best_confidence, 4),
            "agreement_rate": round(agreement_rate, 3),
            "weighted_votes": {k: round(v, 4) for k, v in weighted_votes.items()},
            "tf_signals": tf_signals,
            "reasons": list(set(all_reasons)),
            "divergence": divergence,
        }

        self.signal_history[symbol].append({
            "action": aggregated["action"],
            "confidence": aggregated["confidence"],
            "timestamp": datetime.utcnow().isoformat(),
        })

        return aggregated

    def _detect_divergence(self, tf_signals: dict) -> Optional[str]:
        actions = [s["action"] for s in tf_signals.values() if s["action"] != "HOLD"]
        if len(set(actions)) > 1 and len(actions) >= 3:
            buys = actions.count("BUY")
            sells = actions.count("SELL")
            if buys > 0 and sells > 0:
                return f"TF_DIVERGENCE: {buys}xBUY vs {sells}xSELL"
        return None

    def get_signal_consistency(self, symbol: str) -> float:
        history = list(self.signal_history.get(symbol, []))
        if len(history) < 3:
            return 0.5
        recent = history[-10:]
        actions = [h["action"] for h in recent]
        if not actions:
            return 0.5
        most_common = max(set(actions), key=actions.count)
        return actions.count(most_common) / len(actions)

class PositionTracker:
    def __init__(self):
        self.tracked_positions: Dict[int, dict] = {}
        self.position_events: deque = deque(maxlen=500)
        self.max_position_age_hours: float = 72.0
        self.profit_lock_threshold: float = 0.02
        self.risk_alerts: deque = deque(maxlen=100)

    def update_position(self, position: dict, current_price: float, atr: float):
        ticket = position.get("ticket")
        if ticket is None:
            return

        if ticket not in self.tracked_positions:
            self.tracked_positions[ticket] = {
                "ticket": ticket,
                "symbol": position.get("symbol"),
                "order_type": position.get("order_type"),
                "open_price": position.get("open_price"),
                "volume": position.get("volume"),
                "open_time": position.get("open_time", datetime.utcnow().isoformat()),
                "peak_profit": 0.0,
                "min_profit": 0.0,
                "max_adverse_excursion": 0.0,
                "max_favorable_excursion": 0.0,
                "profit_locked": False,
                "age_hours": 0.0,
                "atr_at_open": atr,
                "updates": 0,
            }

        tp = self.tracked_positions[ticket]
        profit = position.get("profit", 0.0)
        open_price = tp.get("open_price", current_price)
        order_type = tp.get("order_type", "BUY")

        if profit > tp["peak_profit"]:
            tp["peak_profit"] = profit
        if profit < tp["min_profit"]:
            tp["min_profit"] = profit

        if order_type == "BUY":
            adverse = open_price - current_price
            favorable = current_price - open_price
        else:
            adverse = current_price - open_price
            favorable = open_price - current_price

        if adverse > tp["max_adverse_excursion"]:
            tp["max_adverse_excursion"] = adverse
        if favorable > tp["max_favorable_excursion"]:
            tp["max_favorable_excursion"] = favorable

        try:
            open_dt = datetime.fromisoformat(tp["open_time"])
            tp["age_hours"] = (datetime.utcnow() - open_dt).total_seconds() / 3600
        except Exception:
            tp["age_hours"] = 0.0

        balance_proxy = 10000
        if profit > balance_proxy * self.profit_lock_threshold and not tp["profit_locked"]:
            tp["profit_locked"] = True
            self.position_events.append({
                "ticket": ticket,
                "event": "PROFIT_LOCKED",
                "profit": profit,
                "timestamp": datetime.utcnow().isoformat(),
            })

        tp["current_profit"] = profit
        tp["current_price"] = current_price
        tp["updates"] += 1

        self._check_alerts(tp, atr)

    def _check_alerts(self, tp: dict, atr: float):
        ticket = tp.get("ticket")
        age_hours = tp.get("age_hours", 0.0)
        profit = tp.get("current_profit", 0.0)
        peak = tp.get("peak_profit", 0.0)

        if age_hours > self.max_position_age_hours and profit <= 0:
            self.risk_alerts.append({
                "ticket": ticket,
                "type": "AGE_WARNING",
                "message": f"Position {ticket} is {age_hours:.1f}h old with profit={profit:.2f}",
                "timestamp": datetime.utcnow().isoformat(),
            })

        if peak > 0 and profit < peak * 0.5 and peak > atr:
            self.risk_alerts.append({
                "ticket": ticket,
                "type": "PROFIT_PULLBACK",
                "message": f"Profit pulled back from {peak:.2f} to {profit:.2f}",
                "timestamp": datetime.utcnow().isoformat(),
            })

    def remove_position(self, ticket: int) -> Optional[dict]:
        return self.tracked_positions.pop(ticket, None)

    def get_portfolio_heat(self) -> float:
        if not self.tracked_positions:
            return 0.0
        total_adverse = sum(
            tp.get("max_adverse_excursion", 0)
            for tp in self.tracked_positions.values()
        )
        atr_sum = sum(
            tp.get("atr_at_open", 0.001)
            for tp in self.tracked_positions.values()
        )
        if atr_sum <= 0:
            return 0.0
        heat = total_adverse / atr_sum / len(self.tracked_positions)
        return float(np.clip(heat, 0.0, 1.0))

    def get_positions_summary(self) -> dict:
        positions = list(self.tracked_positions.values())
        if not positions:
            return {"count": 0, "total_profit": 0.0, "portfolio_heat": 0.0}
        total_profit = sum(p.get("current_profit", 0) for p in positions)
        return {
            "count": len(positions),
            "total_profit": round(total_profit, 2),
            "portfolio_heat": round(self.get_portfolio_heat(), 3),
            "oldest_hours": round(max(p.get("age_hours", 0) for p in positions), 1),
            "profit_locked_count": sum(1 for p in positions if p.get("profit_locked")),
            "risk_alerts_count": len(self.risk_alerts),
            "positions": [
                {
                    "ticket": p["ticket"],
                    "symbol": p["symbol"],
                    "order_type": p["order_type"],
                    "profit": round(p.get("current_profit", 0), 2),
                    "peak_profit": round(p["peak_profit"], 2),
                    "age_hours": round(p["age_hours"], 2),
                    "mae": round(p["max_adverse_excursion"], 5),
                    "mfe": round(p["max_favorable_excursion"], 5),
                }
                for p in positions
            ],
        }

class MarketRegimeClassifier:
    def __init__(self):
        self.regime_history: Dict[str, deque] = {}
        self.transition_matrix: Dict[str, Dict[str, float]] = {
            r: {r2: 0.0 for r2 in REGIMES} for r in REGIMES
        }
        self.transition_counts: Dict[str, Dict[str, int]] = {
            r: {r2: 0 for r2 in REGIMES} for r in REGIMES
        }
        self.regime_durations: Dict[str, deque] = {r: deque(maxlen=50) for r in REGIMES}
        self.classifier_nn = SimpleNeuralNet(input_size=20, hidden_sizes=[32, 16], output_size=len(REGIMES), lr=0.005)
        self.label_history: deque = deque(maxlen=200)

    def classify(self, symbol: str, candles: dict, indicators: dict) -> Tuple[str, float]:
        try:
            if symbol not in self.regime_history:
                self.regime_history[symbol] = deque(maxlen=100)

            features = self._build_classifier_features(candles, indicators)
            if features is None:
                return "RANGING", 0.5

            nn_output = self.classifier_nn.predict(features.reshape(1, -1))[0]
            nn_probs = self._softmax(nn_output)
            nn_regime_idx = int(np.argmax(nn_probs))
            nn_regime = REGIMES[nn_regime_idx]
            nn_confidence = float(nn_probs[nn_regime_idx])

            stat_regime, stat_confidence = self._statistical_classify(candles, indicators)

            if nn_confidence > 0.7:
                final_regime = nn_regime
                final_confidence = nn_confidence * 0.6 + stat_confidence * 0.4
            else:
                final_regime = stat_regime
                final_confidence = stat_confidence * 0.7 + nn_confidence * 0.3

            history = list(self.regime_history.get(symbol, []))
            if history:
                prev_regime = history[-1]
                trans_prob = self.transition_matrix.get(prev_regime, {}).get(final_regime, 0.1)
                if trans_prob > 0.3:
                    final_confidence = min(0.95, final_confidence * 1.1)
                elif trans_prob < 0.05:
                    final_confidence *= 0.85

            prev = self.regime_history[symbol][-1] if self.regime_history[symbol] else final_regime
            self.regime_history[symbol].append(final_regime)
            self._update_transition_matrix(prev, final_regime)

            return final_regime, round(float(np.clip(final_confidence, 0.3, 0.99)), 3)

        except Exception as e:
            logger.error(f"Regime classification error ({symbol}): {e}")
            return "RANGING", 0.5

    def _build_classifier_features(self, candles: dict, indicators: dict) -> Optional[np.ndarray]:
        try:
            features = []
            m15_ind = indicators.get("m15", {})
            h1_ind = indicators.get("h1", {})
            h4_ind = indicators.get("h4", {})

            close = m15_ind.get("current_close", 0)
            if close <= 0:
                return None

            sma20 = m15_ind.get("sma_20", close)
            sma50 = m15_ind.get("sma_50", close)
            sma200 = h1_ind.get("sma_200", close)
            rsi_m15 = m15_ind.get("rsi", 50)
            rsi_h1 = h1_ind.get("rsi", 50)
            macd_m15 = m15_ind.get("macd", 0)
            atr_m15 = m15_ind.get("atr", close * 0.001)
            bb_upper = m15_ind.get("bollinger_upper", close)
            bb_lower = m15_ind.get("bollinger_lower", close)
            bb_width = (bb_upper - bb_lower) / (close + 1e-10)
            bb_pos = (close - bb_lower) / (bb_upper - bb_lower + 1e-10)

            features.extend([
                (close - sma20) / (sma20 + 1e-10),
                (sma20 - sma50) / (sma50 + 1e-10),
                (close - sma200) / (sma200 + 1e-10),
                (rsi_m15 - 50) / 50,
                (rsi_h1 - 50) / 50,
                macd_m15 / (close + 1e-10) * 1000,
                atr_m15 / (close + 1e-10) * 1000,
                bb_width,
                bb_pos - 0.5,
            ])

            m15_candles = candles.get("m15")
            if m15_candles is not None and len(m15_candles) >= 20:
                closes = np.array([c[4] for c in m15_candles[-20:]])
                returns = np.diff(np.log(closes + 1e-10))
                features.append(float(np.std(returns) * 100))
                features.append(float(np.mean(returns) * 100))
                features.append(float(np.max(closes) - np.min(closes)) / (close + 1e-10))
                features.append(float(closes[-1] - closes[-5]) / (closes[-5] + 1e-10))
                features.append(float(closes[-1] - closes[-10]) / (closes[-10] + 1e-10))
                volumes = np.array([c[5] for c in m15_candles[-20:]])
                avg_vol = float(np.mean(volumes[:-1]))
                features.append(float(volumes[-1]) / (avg_vol + 1e-10) - 1.0)
                body_sizes = np.abs(
                    np.array([c[4] for c in m15_candles[-10:]]) -
                    np.array([c[1] for c in m15_candles[-10:]])
                )
                features.append(float(np.mean(body_sizes)) / (close + 1e-10))
            else:
                features.extend([0.0] * 7)

            h4_ind_data = h4_ind if h4_ind else {}
            rsi_h4 = h4_ind_data.get("rsi", 50)
            macd_h4 = h4_ind_data.get("macd", 0)
            features.append((rsi_h4 - 50) / 50)
            features.append(macd_h4 / (close + 1e-10) * 1000)

            arr = np.array(features[:20], dtype=np.float32)
            if len(arr) < 20:
                arr = np.pad(arr, (0, 20 - len(arr)))
            return np.nan_to_num(np.clip(arr, -5.0, 5.0), nan=0.0)

        except Exception as e:
            logger.error(f"Build classifier features error: {e}")
            return None

    def _statistical_classify(self, candles: dict, indicators: dict) -> Tuple[str, float]:
        try:
            m15 = indicators.get("m15", {})
            close = m15.get("current_close", 0)
            if close <= 0:
                return "RANGING", 0.5

            atr = m15.get("atr", close * 0.001)
            rsi = m15.get("rsi", 50)
            macd = m15.get("macd", 0)
            sma20 = m15.get("sma_20", close)
            sma50 = m15.get("sma_50", close)
            bb_upper = m15.get("bollinger_upper", close)
            bb_lower = m15.get("bollinger_lower", close)
            bb_width = (bb_upper - bb_lower) / (close + 1e-10)

            volatility_score = atr / (close + 1e-10) * 1000
            trend_score = abs(sma20 - sma50) / (sma50 + 1e-10) * 100
            momentum_score = abs(rsi - 50) / 50

            if volatility_score > 3.0:
                return "VOLATILE", min(0.9, 0.5 + volatility_score / 10)

            if close > sma20 > sma50 and macd > 0 and rsi > 52:
                conf = 0.5 + trend_score * 2 + momentum_score * 0.2
                return "TRENDING_UP", min(0.92, conf)

            if close < sma20 < sma50 and macd < 0 and rsi < 48:
                conf = 0.5 + trend_score * 2 + momentum_score * 0.2
                return "TRENDING_DOWN", min(0.92, conf)

            if bb_width < 0.003:
                if close > bb_upper:
                    return "BREAKOUT", 0.75
                if close < bb_lower:
                    return "BREAKOUT", 0.75

            if rsi > 68 and macd < 0:
                return "REVERSAL", 0.70
            if rsi < 32 and macd > 0:
                return "REVERSAL", 0.70

            ranging_conf = 0.5 + (1.0 - trend_score) * 0.3
            return "RANGING", min(0.85, ranging_conf)

        except Exception as e:
            logger.error(f"Statistical classify error: {e}")
            return "RANGING", 0.5

    def _softmax(self, x: np.ndarray) -> np.ndarray:
        e = np.exp(x - np.max(x))
        return e / (e.sum() + 1e-10)

    def _update_transition_matrix(self, prev: str, current: str):
        if prev not in self.transition_counts:
            return
        self.transition_counts[prev][current] = self.transition_counts[prev].get(current, 0) + 1
        total = sum(self.transition_counts[prev].values())
        if total > 0:
            for regime in REGIMES:
                count = self.transition_counts[prev].get(regime, 0)
                self.transition_matrix[prev][regime] = count / total

    def train_on_labeled(self, features: np.ndarray, regime: str):
        if regime not in REGIMES:
            return
        target = np.zeros(len(REGIMES))
        target[REGIMES.index(regime)] = 1.0
        self.classifier_nn.train_step(features.reshape(1, -1), target.reshape(1, -1))
        self.label_history.append({"features": features.tolist(), "regime": regime})

    def get_regime_probabilities(self, symbol: str) -> Dict[str, float]:
        history = list(self.regime_history.get(symbol, []))
        if not history:
            return {r: 1.0 / len(REGIMES) for r in REGIMES}
        counts = {r: history.count(r) for r in REGIMES}
        total = len(history)
        return {r: round(counts[r] / total, 3) for r in REGIMES}

    def get_transition_stats(self) -> dict:
        return {
            "transition_matrix": {
                r: {r2: round(v, 3) for r2, v in row.items()}
                for r, row in self.transition_matrix.items()
            },
            "transition_counts": self.transition_counts,
        }

# EVENT BUS

@dataclass
class TradingEvent:
    event_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    event_type: str = ""
    source: str = ""
    payload: Dict[str, Any] = field(default_factory=dict)
    timestamp: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    priority: int = 1

EventHandler = Callable[[TradingEvent], Coroutine[Any, Any, None]]

class EventBus:
    TRADE_OPENED       = "trade_opened"
    TRADE_CLOSED       = "trade_closed"
    TRADE_MODIFIED     = "trade_modified"
    REGIME_CHANGED     = "regime_changed"
    RISK_ALERT         = "risk_alert"
    SIGNAL_GENERATED   = "signal_generated"
    DRAWDOWN_BREACH    = "drawdown_breach"
    KILL_SWITCH        = "kill_switch"
    MODEL_DRIFT        = "model_drift"
    PERFORMANCE_ALERT  = "performance_alert"
    LEARNING_UPDATED   = "learning_updated"
    EXPOSURE_BREACH    = "exposure_breach"
    OVERTRADING_ALERT  = "overtrading_alert"
    SYSTEM_READY       = "system_ready"

    def __init__(self):
        self._subscribers: Dict[str, List[EventHandler]] = {}
        self._event_queue: asyncio.Queue = asyncio.Queue(maxsize=2000)
        self._event_log: deque = deque(maxlen=1000)
        self._handler_errors: deque = deque(maxlen=200)
        self._running = False
        self._processor_task: Optional[asyncio.Task] = None
        self._stats: Dict[str, int] = {}
        logger.info("EventBus initialized")

    def subscribe(self, event_type: str, handler: EventHandler):
        if event_type not in self._subscribers:
            self._subscribers[event_type] = []
        if handler not in self._subscribers[event_type]:
            self._subscribers[event_type].append(handler)

    def unsubscribe(self, event_type: str, handler: EventHandler):
        if event_type in self._subscribers:
            self._subscribers[event_type] = [
                h for h in self._subscribers[event_type] if h != handler
            ]

    async def publish(self, event: TradingEvent):
        try:
            self._event_log.append({
                "id": event.event_id,
                "type": event.event_type,
                "source": event.source,
                "timestamp": event.timestamp,
                "priority": event.priority,
            })
            self._stats[event.event_type] = self._stats.get(event.event_type, 0) + 1
            if not self._event_queue.full():
                await self._event_queue.put(event)
            else:
                logger.warning(f"EventBus queue full — dropping event {event.event_type}")
        except Exception as e:
            logger.error(f"EventBus publish error: {e}")

    async def publish_sync(self, event: TradingEvent):
        await self._dispatch(event)

    async def _dispatch(self, event: TradingEvent):
        handlers = self._subscribers.get(event.event_type, [])
        wildcard = self._subscribers.get("*", [])
        all_handlers = handlers + [h for h in wildcard if h not in handlers]
        for handler in all_handlers:
            try:
                await handler(event)
            except Exception as e:
                self._handler_errors.append({
                    "handler": getattr(handler, "__name__", "unknown"),
                    "event_type": event.event_type,
                    "error": str(e),
                    "timestamp": datetime.utcnow().isoformat(),
                })
                logger.error(f"EventBus handler {handler.__name__} error on {event.event_type}: {e}")

    async def start(self):
        self._running = True
        self._processor_task = asyncio.create_task(self._process_loop())
        logger.info("EventBus processor started")

    async def stop(self):
        self._running = False
        if self._processor_task and not self._processor_task.done():
            self._processor_task.cancel()
            try:
                await self._processor_task
            except asyncio.CancelledError:
                pass
        logger.info("EventBus processor stopped")

    async def _process_loop(self):
        while self._running:
            try:
                try:
                    event = await asyncio.wait_for(self._event_queue.get(), timeout=1.0)
                    await self._dispatch(event)
                    self._event_queue.task_done()
                except asyncio.TimeoutError:
                    continue
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"EventBus process loop error: {e}")
                await asyncio.sleep(0.1)

    def get_stats(self) -> dict:
        return {
            "queue_size": self._event_queue.qsize(),
            "total_events": sum(self._stats.values()),
            "events_by_type": dict(self._stats),
            "subscribers": {k: len(v) for k, v in self._subscribers.items()},
            "handler_errors": len(self._handler_errors),
            "recent_errors": list(self._handler_errors)[-5:],
            "recent_events": list(self._event_log)[-10:],
        }

class AdvancedRiskEngine:
    def __init__(self, event_bus: Optional["EventBus"] = None):
        self.event_bus = event_bus
        self.portfolio_returns: deque = deque(maxlen=500)
        self.symbol_returns: Dict[str, deque] = {}
        self.correlation_matrix: Dict[str, Dict[str, float]] = {}
        self.correlation_last_update: Optional[datetime] = None
        self.correlation_update_interval = 300

        self.var_confidence = 0.95
        self.var_horizon_days = 1
        self.cvar_window = 252

        self.gross_exposure: float = 0.0
        self.net_exposure: float = 0.0
        self.symbol_exposure: Dict[str, float] = {}
        self.sector_exposure: Dict[str, float] = {}
        self.max_gross_exposure_pct: float = 0.30
        self.max_symbol_exposure_pct: float = 0.10
        self.max_sector_exposure_pct: float = 0.20

        self.current_leverage: float = 1.0
        self.max_leverage: float = 5.0
        self.target_volatility: float = 0.15
        self.leverage_history: deque = deque(maxlen=100)

        self.portfolio_risk_score: float = 0.0
        self.last_var: float = 0.0
        self.last_cvar: float = 0.0
        self.last_score_update: Optional[datetime] = None

        self.stress_scenarios: Dict[str, float] = {
            "flash_crash_10pct": -0.10,
            "flash_crash_5pct": -0.05,
            "gap_up_5pct": 0.05,
            "volatility_spike": -0.07,
        }

        self.symbol_sectors: Dict[str, str] = {
            "EURUSD": "FX_MAJOR", "GBPUSD": "FX_MAJOR",
            "USDJPY": "FX_MAJOR", "USDCHF": "FX_MAJOR",
            "AUDUSD": "FX_COMMODITY", "NZDUSD": "FX_COMMODITY",
            "USDCAD": "FX_COMMODITY", "XAUUSD": "METALS",
            "XAGUSD": "METALS", "BTCUSD": "CRYPTO",
            "ETHUSD": "CRYPTO", "US30": "INDICES",
            "SPX500": "INDICES", "NAS100": "INDICES",
        }
        logger.info("AdvancedRiskEngine initialized")

    def record_symbol_return(self, symbol: str, ret: float):
        if symbol not in self.symbol_returns:
            self.symbol_returns[symbol] = deque(maxlen=252)
        self.symbol_returns[symbol].append(ret)
        if self.symbol_returns:
            avg_ret = float(np.mean([
                list(rets)[-1]
                for rets in self.symbol_returns.values()
                if rets
            ]))
            self.portfolio_returns.append(avg_ret)

    def compute_var(self, confidence: float = 0.95, horizon: int = 1) -> Tuple[float, float]:
        try:
            returns = list(self.portfolio_returns)
            if len(returns) < 20:
                return 0.02, 0.03
            arr = np.array(returns)
            scaled = arr * np.sqrt(horizon)
            var_pct = float(np.percentile(scaled, (1 - confidence) * 100))
            tail = scaled[scaled <= var_pct]
            cvar_pct = float(np.mean(tail)) if len(tail) > 0 else var_pct
            self.last_var = abs(var_pct)
            self.last_cvar = abs(cvar_pct)
            return self.last_var, self.last_cvar
        except Exception as e:
            logger.error(f"VaR computation error: {e}")
            return 0.02, 0.03

    def compute_portfolio_volatility(self) -> float:
        try:
            returns = list(self.portfolio_returns)
            if len(returns) < 5:
                return 0.15
            arr = np.array(returns)
            daily_vol = float(np.std(arr))
            ann_vol = daily_vol * np.sqrt(252)
            return float(np.clip(ann_vol, 0.01, 5.0))
        except Exception as e:
            logger.error(f"Portfolio volatility error: {e}")
            return 0.15

    def compute_correlation_matrix(self) -> Dict[str, Dict[str, float]]:
        try:
            now = datetime.utcnow()
            if (
                self.correlation_last_update
                and (now - self.correlation_last_update).total_seconds() < self.correlation_update_interval
            ):
                return self.correlation_matrix

            symbols = [s for s, r in self.symbol_returns.items() if len(r) >= 20]
            if len(symbols) < 2:
                return {}

            matrix: Dict[str, Dict[str, float]] = {}
            for s1 in symbols:
                matrix[s1] = {}
                r1 = np.array(list(self.symbol_returns[s1]))
                for s2 in symbols:
                    r2 = np.array(list(self.symbol_returns[s2]))
                    min_len = min(len(r1), len(r2))
                    if min_len < 10:
                        matrix[s1][s2] = 0.0
                        continue
                    corr = float(np.corrcoef(r1[-min_len:], r2[-min_len:])[0, 1])
                    matrix[s1][s2] = round(float(np.nan_to_num(corr, nan=0.0)), 3)

            self.correlation_matrix = matrix
            self.correlation_last_update = now
            return matrix
        except Exception as e:
            logger.error(f"Correlation matrix error: {e}")
            return {}

    def get_correlation_risk(self, symbol: str, existing_positions: list) -> float:
        try:
            if not existing_positions:
                return 0.0
            matrix = self.compute_correlation_matrix()
            if not matrix or symbol not in matrix:
                return 0.0
            open_symbols = list(set(p.get("symbol") for p in existing_positions))
            correlations = []
            for s in open_symbols:
                if s in matrix.get(symbol, {}):
                    correlations.append(abs(matrix[symbol][s]))
            if not correlations:
                return 0.0
            max_corr = float(np.max(correlations))
            avg_corr = float(np.mean(correlations))
            return float(np.clip(max_corr * 0.7 + avg_corr * 0.3, 0.0, 1.0))
        except Exception as e:
            logger.error(f"Correlation risk error: {e}")
            return 0.0

    def compute_dynamic_leverage(self) -> float:
        try:
            port_vol = self.compute_portfolio_volatility()
            if port_vol <= 0:
                return 1.0
            target_lev = self.target_volatility / port_vol
            var, cvar = self.compute_var()
            if cvar > 0.05:
                target_lev *= 0.5
            elif cvar > 0.03:
                target_lev *= 0.75
            alpha = 0.2
            self.current_leverage = (
                alpha * float(np.clip(target_lev, 0.1, self.max_leverage))
                + (1 - alpha) * self.current_leverage
            )
            self.leverage_history.append({
                "leverage": self.current_leverage,
                "port_vol": port_vol,
                "timestamp": datetime.utcnow().isoformat(),
            })
            return round(self.current_leverage, 3)
        except Exception as e:
            logger.error(f"Dynamic leverage error: {e}")
            return 1.0

    def update_exposure(self, positions: list, account: dict):
        try:
            balance = account.get("balance", 10000)
            if balance <= 0:
                return
            self.symbol_exposure.clear()
            self.sector_exposure.clear()
            gross = 0.0
            net = 0.0
            for p in positions:
                symbol = p.get("symbol", "UNKNOWN")
                volume = p.get("volume", 0.0)
                open_price = p.get("open_price", 1.0)
                order_type = p.get("order_type", "BUY")
                notional = volume * open_price * 100000
                exposure_pct = notional / balance
                self.symbol_exposure[symbol] = (
                    self.symbol_exposure.get(symbol, 0.0) + exposure_pct
                )
                sector = self.symbol_sectors.get(symbol, "OTHER")
                self.sector_exposure[sector] = (
                    self.sector_exposure.get(sector, 0.0) + exposure_pct
                )
                gross += exposure_pct
                net += exposure_pct if order_type == "BUY" else -exposure_pct
            self.gross_exposure = gross
            self.net_exposure = net
        except Exception as e:
            logger.error(f"Update exposure error: {e}")

    def run_stress_test(self, positions: list, account: dict) -> Dict[str, float]:
        try:
            balance = account.get("balance", 10000)
            if balance <= 0:
                return {}
            results = {}
            for scenario_name, shock in self.stress_scenarios.items():
                total_pnl = 0.0
                for p in positions:
                    volume = p.get("volume", 0.0)
                    open_price = p.get("open_price", 1.0)
                    order_type = p.get("order_type", "BUY")
                    notional = volume * open_price * 100000
                    if order_type == "BUY":
                        pnl = notional * shock
                    else:
                        pnl = notional * (-shock)
                    total_pnl += pnl
                results[scenario_name] = round(total_pnl / balance, 4)
            return results
        except Exception as e:
            logger.error(f"Stress test error: {e}")
            return {}

    def compute_portfolio_risk_score(
        self,
        positions: list,
        account: dict,
    ) -> float:
        try:
            balance = account.get("balance", 10000)
            equity = account.get("equity", balance)
            if balance <= 0:
                return 0.0

            self.update_exposure(positions, account)
            var, cvar = self.compute_var()
            port_vol = self.compute_portfolio_volatility()
            drawdown = (balance - equity) / balance if balance > 0 else 0.0

            var_score = float(np.clip(var / 0.05, 0.0, 1.0))
            cvar_score = float(np.clip(cvar / 0.08, 0.0, 1.0))
            exposure_score = float(np.clip(self.gross_exposure / self.max_gross_exposure_pct, 0.0, 1.0))
            vol_score = float(np.clip(port_vol / 0.50, 0.0, 1.0))
            dd_score = float(np.clip(drawdown / 0.15, 0.0, 1.0))

            if len(positions) >= 2:
                sym_list = list(set(p.get("symbol") for p in positions))
                max_corr = 0.0
                matrix = self.compute_correlation_matrix()
                for i, s1 in enumerate(sym_list):
                    for s2 in sym_list[i+1:]:
                        c = abs(matrix.get(s1, {}).get(s2, 0.0))
                        max_corr = max(max_corr, c)
                corr_score = float(np.clip(max_corr, 0.0, 1.0))
            else:
                corr_score = 0.0

            weights = {
                "var": 0.20,
                "cvar": 0.25,
                "exposure": 0.20,
                "vol": 0.15,
                "dd": 0.15,
                "corr": 0.05,
            }
            score = (
                weights["var"] * var_score
                + weights["cvar"] * cvar_score
                + weights["exposure"] * exposure_score
                + weights["vol"] * vol_score
                + weights["dd"] * dd_score
                + weights["corr"] * corr_score
            )
            self.portfolio_risk_score = float(np.clip(score, 0.0, 1.0))
            self.last_score_update = datetime.utcnow()
            return self.portfolio_risk_score
        except Exception as e:
            logger.error(f"Portfolio risk score error: {e}")
            return 0.5

    def suggest_position_size(
        self,
        symbol: str,
        account: dict,
        positions: list,
        entry_price: float,
        sl_price: float,
        base_volume: float,
    ) -> Tuple[float, str]:
        try:
            risk_score = self.compute_portfolio_risk_score(positions, account)
            corr_risk = self.get_correlation_risk(symbol, positions)
            leverage = self.compute_dynamic_leverage()

            multiplier = 1.0
            reasons = []

            if risk_score > 0.80:
                multiplier *= 0.25
                reasons.append(f"CRITICAL risk_score={risk_score:.2f}")
            elif risk_score > 0.65:
                multiplier *= 0.50
                reasons.append(f"HIGH risk_score={risk_score:.2f}")
            elif risk_score > 0.50:
                multiplier *= 0.75
                reasons.append(f"ELEVATED risk_score={risk_score:.2f}")

            if corr_risk > 0.80:
                multiplier *= 0.50
                reasons.append(f"HIGH correlation={corr_risk:.2f}")
            elif corr_risk > 0.60:
                multiplier *= 0.75
                reasons.append(f"MODERATE correlation={corr_risk:.2f}")

            symbol_exp = self.symbol_exposure.get(symbol, 0.0)
            if symbol_exp > self.max_symbol_exposure_pct:
                multiplier *= 0.30
                reasons.append(f"Symbol exposure cap ({symbol_exp:.1%})")

            sector = self.symbol_sectors.get(symbol, "OTHER")
            sector_exp = self.sector_exposure.get(sector, 0.0)
            if sector_exp > self.max_sector_exposure_pct:
                multiplier *= 0.50
                reasons.append(f"Sector exposure cap ({sector_exp:.1%})")

            multiplier *= float(np.clip(leverage / self.max_leverage, 0.2, 1.5))

            adjusted = round(float(np.clip(base_volume * multiplier, 0.01, 10.0)), 2)
            reason_str = " | ".join(reasons) if reasons else "OK"
            return adjusted, reason_str

        except Exception as e:
            logger.error(f"Suggest position size error: {e}")
            return base_volume, "error"

    async def emit_risk_alerts(self, event_bus: "EventBus", positions: list, account: dict):
        try:
            score = self.compute_portfolio_risk_score(positions, account)
            var, cvar = self.compute_var()

            if score > 0.80:
                await event_bus.publish(TradingEvent(
                    event_type=EventBus.RISK_ALERT,
                    source="AdvancedRiskEngine",
                    payload={
                        "type": "CRITICAL_RISK_SCORE",
                        "score": score,
                        "var": var,
                        "cvar": cvar,
                        "gross_exposure": self.gross_exposure,
                    },
                    priority=3,
                ))

            if self.gross_exposure > self.max_gross_exposure_pct:
                await event_bus.publish(TradingEvent(
                    event_type=EventBus.EXPOSURE_BREACH,
                    source="AdvancedRiskEngine",
                    payload={
                        "gross_exposure": self.gross_exposure,
                        "limit": self.max_gross_exposure_pct,
                    },
                    priority=3,
                ))

            balance = account.get("balance", 10000)
            equity = account.get("equity", balance)
            if balance > 0:
                dd = (balance - equity) / balance
                if dd > 0.10:
                    await event_bus.publish(TradingEvent(
                        event_type=EventBus.DRAWDOWN_BREACH,
                        source="AdvancedRiskEngine",
                        payload={"drawdown": dd, "balance": balance, "equity": equity},
                        priority=3,
                    ))
        except Exception as e:
            logger.error(f"Emit risk alerts error: {e}")

    def get_full_risk_report(self) -> dict:
        var, cvar = self.compute_var()
        port_vol = self.compute_portfolio_volatility()
        leverage = self.current_leverage
        return {
            "portfolio_risk_score": round(self.portfolio_risk_score, 4),
            "var_95_1d": round(var, 4),
            "cvar_95_1d": round(cvar, 4),
            "portfolio_volatility_ann": round(port_vol, 4),
            "current_leverage": round(leverage, 3),
            "gross_exposure": round(self.gross_exposure, 4),
            "net_exposure": round(self.net_exposure, 4),
            "symbol_exposure": {k: round(v, 4) for k, v in self.symbol_exposure.items()},
            "sector_exposure": {k: round(v, 4) for k, v in self.sector_exposure.items()},
            "last_score_update": self.last_score_update.isoformat() if self.last_score_update else None,
            "correlation_matrix": {
                s1: {s2: v for s2, v in row.items()}
                for s1, row in self.correlation_matrix.items()
            },
        }

@dataclass
class FeatureRecord:
    timestamp: str
    symbol: str
    regime: str
    strategy: str
    features: List[float]
    label: int
    confidence: float
    profit: float = 0.0

class OnlineLearningEngine:
    def __init__(self, feature_dim: int = 50, event_bus: Optional["EventBus"] = None):
        self.feature_dim = feature_dim
        self.event_bus = event_bus

        self.feature_store: deque = deque(maxlen=5000)
        self.regime_feature_store: Dict[str, deque] = {
            r: deque(maxlen=500) for r in REGIMES
        }
        self.strategy_feature_store: Dict[str, deque] = {
            s: deque(maxlen=500) for s in STRATEGIES
        }

        self.regime_classifiers: Dict[str, SimpleNeuralNet] = {
            r: SimpleNeuralNet(feature_dim, [64, 32], 2, lr=0.002) for r in REGIMES
        }
        self.strategy_classifiers: Dict[str, SimpleNeuralNet] = {
            s: SimpleNeuralNet(feature_dim, [64, 32], 2, lr=0.002) for s in STRATEGIES
        }
        self.global_classifier = SimpleNeuralNet(feature_dim, [128, 64, 32], 2, lr=0.001)

        self.baseline_accuracy: Dict[str, float] = {}
        self.rolling_accuracy: Dict[str, deque] = {}
        self.drift_threshold = 0.15
        self.drift_window = 50
        self.drift_events: deque = deque(maxlen=100)
        self.degradation_alerts: deque = deque(maxlen=50)

        self.accuracy_window: deque = deque(maxlen=200)
        self.regime_accuracy: Dict[str, deque] = {r: deque(maxlen=100) for r in REGIMES}
        self.strategy_accuracy: Dict[str, deque] = {s: deque(maxlen=100) for s in STRATEGIES}

        self.pending_train: deque = deque(maxlen=1000)
        self.total_online_updates = 0
        self.last_drift_check: Optional[datetime] = None

        logger.info("OnlineLearningEngine initialized")

    def store_feature(self, record: FeatureRecord):
        self.feature_store.append(record)
        self.regime_feature_store[record.regime].append(record)
        self.strategy_feature_store.get(record.strategy, deque()).append(record)
        self.pending_train.append(record)

    def online_update(self, record: FeatureRecord) -> float:
        try:
            features = np.array(record.features[:self.feature_dim], dtype=np.float32)
            if len(features) < self.feature_dim:
                features = np.pad(features, (0, self.feature_dim - len(features)))
            features = np.nan_to_num(np.clip(features, -5.0, 5.0))

            target = np.array([[float(record.label), float(1 - record.label)]], dtype=np.float32)

            loss = self.global_classifier.train_step(features.reshape(1, -1), target)

            if record.regime in self.regime_classifiers:
                loss += self.regime_classifiers[record.regime].train_step(
                    features.reshape(1, -1), target
                )

            if record.strategy in self.strategy_classifiers:
                loss += self.strategy_classifiers[record.strategy].train_step(
                    features.reshape(1, -1), target
                )

            pred = self.global_classifier.predict(features.reshape(1, -1))[0]
            predicted_label = int(np.argmax(pred))
            correct = int(predicted_label == record.label)

            self.accuracy_window.append(correct)
            self.regime_accuracy[record.regime].append(correct)
            if record.strategy in self.strategy_accuracy:
                self.strategy_accuracy[record.strategy].append(correct)

            self.total_online_updates += 1
            return float(loss)

        except Exception as e:
            logger.error(f"Online update error: {e}")
            return 0.0

    def batch_train(self, n: int = 64) -> Dict[str, float]:
        results = {}
        try:
            records = list(self.pending_train)
            if len(records) < 8:
                return results

            batch = random.sample(records, min(n, len(records)))
            total_loss = 0.0
            for rec in batch:
                loss = self.online_update(rec)
                total_loss += loss

            self.pending_train.clear()
            results["batch_loss"] = round(total_loss / len(batch), 6)
            results["batch_size"] = len(batch)
        except Exception as e:
            logger.error(f"Batch train error: {e}")
        return results

    def predict_trade_quality(
        self,
        features: np.ndarray,
        regime: str,
        strategy: str,
    ) -> Tuple[float, float]:
        try:
            f = np.array(features[:self.feature_dim], dtype=np.float32)
            if len(f) < self.feature_dim:
                f = np.pad(f, (0, self.feature_dim - len(f)))
            f = np.nan_to_num(np.clip(f, -5.0, 5.0)).reshape(1, -1)

            global_pred = self.global_classifier.predict(f)[0]
            global_win_prob = float(global_pred[0])

            regime_win_prob = 0.5
            if regime in self.regime_classifiers:
                regime_pred = self.regime_classifiers[regime].predict(f)[0]
                regime_win_prob = float(regime_pred[0])

            strategy_win_prob = 0.5
            if strategy in self.strategy_classifiers:
                strat_pred = self.strategy_classifiers[strategy].predict(f)[0]
                strategy_win_prob = float(strat_pred[0])

            acc_global = self._get_accuracy("global")
            acc_regime = self._get_accuracy(f"regime_{regime}")
            acc_strategy = self._get_accuracy(f"strategy_{strategy}")
            total_acc = acc_global + acc_regime + acc_strategy + 1e-10

            win_prob = (
                acc_global * global_win_prob
                + acc_regime * regime_win_prob
                + acc_strategy * strategy_win_prob
            ) / total_acc

            confidence = abs(win_prob - 0.5) * 2.0
            return float(np.clip(win_prob, 0.0, 1.0)), float(np.clip(confidence, 0.0, 1.0))

        except Exception as e:
            logger.error(f"Predict trade quality error: {e}")
            return 0.5, 0.0

    def _get_accuracy(self, key: str) -> float:
        if key == "global":
            window = list(self.accuracy_window)
        elif key.startswith("regime_"):
            regime = key[7:]
            window = list(self.regime_accuracy.get(regime, []))
        elif key.startswith("strategy_"):
            strategy = key[9:]
            window = list(self.strategy_accuracy.get(strategy, []))
        else:
            return 0.5
        if not window:
            return 0.5
        return float(np.mean(window[-self.drift_window:]))

    async def check_model_drift(self) -> Optional[dict]:
        try:
            now = datetime.utcnow()
            if (
                self.last_drift_check
                and (now - self.last_drift_check).total_seconds() < 300
            ):
                return None
            self.last_drift_check = now

            drift_detected = False
            drift_details = {}

            window = list(self.accuracy_window)
            if len(window) >= self.drift_window * 2:
                old_acc = float(np.mean(window[: self.drift_window]))
                new_acc = float(np.mean(window[-self.drift_window:]))
                key = "global"
                if key not in self.baseline_accuracy:
                    self.baseline_accuracy[key] = old_acc
                baseline = self.baseline_accuracy[key]
                drop = baseline - new_acc
                if drop > self.drift_threshold:
                    drift_detected = True
                    drift_details["global"] = {
                        "baseline": round(baseline, 3),
                        "current": round(new_acc, 3),
                        "drop": round(drop, 3),
                    }

            for regime, acc_deque in self.regime_accuracy.items():
                w = list(acc_deque)
                if len(w) >= 20:
                    key = f"regime_{regime}"
                    curr_acc = float(np.mean(w[-20:]))
                    if key not in self.baseline_accuracy:
                        self.baseline_accuracy[key] = curr_acc
                    baseline = self.baseline_accuracy[key]
                    drop = baseline - curr_acc
                    if drop > self.drift_threshold * 1.2:
                        drift_detected = True
                        drift_details[key] = {
                            "baseline": round(baseline, 3),
                            "current": round(curr_acc, 3),
                            "drop": round(drop, 3),
                        }

            if drift_detected:
                report = {
                    "drift_detected": True,
                    "details": drift_details,
                    "timestamp": now.isoformat(),
                    "total_updates": self.total_online_updates,
                }
                self.drift_events.append(report)

                if self.event_bus:
                    await self.event_bus.publish(TradingEvent(
                        event_type=EventBus.MODEL_DRIFT,
                        source="OnlineLearningEngine",
                        payload=report,
                        priority=2,
                    ))

                for key in drift_details:
                    if key.startswith("regime_"):
                        regime = key[7:]
                        w = list(self.regime_accuracy.get(regime, []))
                        if w:
                            self.baseline_accuracy[key] = float(np.mean(w[-10:]))
                    else:
                        w = list(self.accuracy_window)
                        if w:
                            self.baseline_accuracy["global"] = float(np.mean(w[-20:]))

                return report

            return None

        except Exception as e:
            logger.error(f"Check model drift error: {e}")
            return None

    async def check_performance_degradation(self, memory_engine: "MemoryEngine") -> Optional[dict]:
        try:
            summary = memory_engine.get_recent_performance(50)
            trend = summary.get("trend", "NEUTRAL")
            avg_reward = summary.get("avg_reward", 0.0)

            if trend == "DECLINING" and avg_reward < -0.3:
                alert = {
                    "type": "PERFORMANCE_DEGRADATION",
                    "trend": trend,
                    "avg_reward": avg_reward,
                    "timestamp": datetime.utcnow().isoformat(),
                }
                self.degradation_alerts.append(alert)
                if self.event_bus:
                    await self.event_bus.publish(TradingEvent(
                        event_type=EventBus.PERFORMANCE_ALERT,
                        source="OnlineLearningEngine",
                        payload=alert,
                        priority=2,
                    ))
                return alert
            return None
        except Exception as e:
            logger.error(f"Performance degradation check error: {e}")
            return None

    def get_feature_store_stats(self) -> dict:
        return {
            "total_records": len(self.feature_store),
            "pending_train": len(self.pending_train),
            "global_accuracy": round(self._get_accuracy("global"), 3),
            "regime_accuracy": {
                r: round(self._get_accuracy(f"regime_{r}"), 3) for r in REGIMES
            },
            "strategy_accuracy": {
                s: round(self._get_accuracy(f"strategy_{s}"), 3) for s in STRATEGIES
            },
            "total_online_updates": self.total_online_updates,
            "drift_events": len(self.drift_events),
            "degradation_alerts": len(self.degradation_alerts),
            "recent_drift": list(self.drift_events)[-3:],
        }

class PersistenceLayer:
    SCHEMA_VERSION = "1.2"

    def __init__(self, base_dir: str = ".", db_path: str = "trading.db"):
        self.base_dir = base_dir
        self.db_path = db_path
        self.state_dir = os.path.join(base_dir, "ai_state")
        os.makedirs(self.state_dir, exist_ok=True)
        self._init_persistence_db()
        logger.info(f"PersistenceLayer initialized at {self.state_dir}")

    def _init_persistence_db(self):
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                CREATE TABLE IF NOT EXISTS ai_state_blobs (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    component TEXT NOT NULL,
                    version TEXT NOT NULL,
                    data_json TEXT,
                    saved_at TEXT,
                    schema_version TEXT
                )
            """)
            c.execute("""
                CREATE TABLE IF NOT EXISTS persistence_log (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp TEXT,
                    operation TEXT,
                    component TEXT,
                    success INTEGER,
                    message TEXT
                )
            """)
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Init persistence DB error: {e}")

    def _log_operation(self, operation: str, component: str, success: bool, message: str = ""):
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute(
                "INSERT INTO persistence_log (timestamp, operation, component, success, message) "
                "VALUES (?,?,?,?,?)",
                (datetime.utcnow().isoformat(), operation, component, int(success), message)
            )
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Persistence log error: {e}")

        except Exception as e:
            logger.error(f"Save regime classifier error: {e}")
            self._log_operation("SAVE", "regime_classifier", False, str(e))

    def load_regime_classifier(self, classifier: "MarketRegimeClassifier") -> bool:
        try:
            data = self._load_json("regime_classifier.json.gz", compressed=True)
            if data is None:
                return False

            counts = data.get("transition_counts", {})
            for r in REGIMES:
                if r in counts:
                    for r2 in REGIMES:
                        classifier.transition_counts[r][r2] = counts[r].get(r2, 0)

            matrix = data.get("transition_matrix", {})
            for r in REGIMES:
                if r in matrix:
                    for r2 in REGIMES:
                        classifier.transition_matrix[r][r2] = matrix[r].get(r2, 0.0)

            for sym, hist in data.get("regime_history", {}).items():
                if sym not in classifier.regime_history:
                    classifier.regime_history[sym] = deque(maxlen=100)
                for item in hist:
                    classifier.regime_history[sym].append(item)

            nn_weights = data.get("nn_weights")
            if nn_weights:
                try:
                    classifier.classifier_nn.set_weights(nn_weights)
                except Exception as we:
                    logger.warning(f"Could not restore regime classifier NN weights: {we}")

            self._log_operation("LOAD", "regime_classifier", True)
            logger.info("Regime classifier state restored")
            return True
        except Exception as e:
            logger.error(f"Load regime classifier error: {e}")

    def _save_json(self, data: dict, filename: str, compress: bool = False):
        path = os.path.join(self.state_dir, filename)
        payload = json.dumps(data, default=str, ensure_ascii=False)
        if compress:
            with gzip.open(path, "wt", encoding="utf-8") as f:
                f.write(payload)
        else:
            with open(path, "w", encoding="utf-8") as f:
                f.write(payload)

    def _load_json(self, filename: str, compressed: bool = False) -> Optional[dict]:
        path = os.path.join(self.state_dir, filename)
        if not os.path.exists(path):
            return None
        try:
            if compressed:
                with gzip.open(path, "rt", encoding="utf-8") as f:
                    return json.loads(f.read())
            else:
                with open(path, "r", encoding="utf-8") as f:
                    return json.loads(f.read())
        except Exception as e:
            logger.error(f"Load JSON {filename} error: {e}")
            return None

    def _save_to_db(self, component: str, data: dict):
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO ai_state_blobs (component, version, data_json, saved_at, schema_version)
                VALUES (?, ?, ?, ?, ?)
            """, (
                component,
                self.SCHEMA_VERSION,
                json.dumps(data, default=str, ensure_ascii=False),
                datetime.utcnow().isoformat(),
                self.SCHEMA_VERSION,
            ))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Save to DB error ({component}): {e}")

    def _load_from_db(self, component: str) -> Optional[dict]:
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                SELECT data_json FROM ai_state_blobs
                WHERE component = ?
                ORDER BY id DESC LIMIT 1
            """, (component,))
            row = c.fetchone()
            conn.close()
            if row and row[0]:
                return json.loads(row[0])
            return None
        except Exception as e:
            logger.error(f"Load from DB error ({component}): {e}")
            return None

    def list_saved_states(self) -> List[dict]:

# ─────────────────────────────────────────────
# EXECUTION SAFETY LAYER
# ─────────────────────────────────────────────
@dataclass
class CooldownEntry:
    symbol: str
    triggered_at: datetime
    duration_seconds: float
    reason: str

class ExecutionSafetyLayer:
    def __init__(self, event_bus: Optional["EventBus"] = None):
        self.event_bus = event_bus

        self.kill_switch_active: bool = False
        self.kill_switch_reason: str = ""
        self.kill_switch_triggered_at: Optional[datetime] = None
        self.kill_switch_auto_reset_seconds: float = 3600.0

        self.symbol_cooldowns: Dict[str, CooldownEntry] = {}
        self.default_cooldown_seconds: float = 60.0
        self.cooldown_after_loss_seconds: float = 300.0
        self.cooldown_after_win_seconds: float = 30.0

        self.trade_timestamps: deque = deque(maxlen=500)
        self.max_trades_per_hour: int = 20
        self.max_trades_per_minute: int = 3
        self.overtrading_window_seconds: float = 3600.0

        self.recent_signals: Dict[str, dict] = {}
        self.signal_dedup_window_seconds: float = 30.0

        self.open_directions: Dict[str, str] = {}

        self.max_concurrent_positions: int = 10
        self.max_positions_per_symbol: int = 2
        self.max_daily_trades: int = 50
        self.daily_trade_count: int = 0
        self.daily_trade_reset: datetime = datetime.utcnow()

        self.blocked_symbols: set = set()

        self.blocked_log: deque = deque(maxlen=500)
        self.allowed_log: deque = deque(maxlen=200)

        logger.info("ExecutionSafetyLayer initialized")

    async def activate_kill_switch(self, reason: str):
        self.kill_switch_active = True
        self.kill_switch_reason = reason
        self.kill_switch_triggered_at = datetime.utcnow()
        logger.critical(f"KILL-SWITCH ACTIVATED: {reason}")
        if self.event_bus:
            await self.event_bus.publish_sync(TradingEvent(
                event_type=EventBus.KILL_SWITCH,
                source="ExecutionSafetyLayer",
                payload={
                    "reason": reason,
                    "triggered_at": self.kill_switch_triggered_at.isoformat(),
                },
                priority=3,
            ))

    def deactivate_kill_switch(self, manual: bool = False) -> bool:
        if not self.kill_switch_active:
            return False
        if not manual:
            if self.kill_switch_triggered_at:
                elapsed = (datetime.utcnow() - self.kill_switch_triggered_at).total_seconds()
                if elapsed < self.kill_switch_auto_reset_seconds:
                    return False
        self.kill_switch_active = False
        self.kill_switch_reason = ""
        self.kill_switch_triggered_at = None
        logger.info(f"Kill-switch deactivated (manual={manual})")
        return True

    def check_kill_switch_auto_reset(self):
        if (
            self.kill_switch_active
            and self.kill_switch_triggered_at
        ):
            elapsed = (datetime.utcnow() - self.kill_switch_triggered_at).total_seconds()
            if elapsed >= self.kill_switch_auto_reset_seconds:
                self.deactivate_kill_switch(manual=False)

    def set_cooldown(
        self,
        symbol: str,
        duration_seconds: float,
        reason: str = "default",
    ):
        self.symbol_cooldowns[symbol] = CooldownEntry(
            symbol=symbol,
            triggered_at=datetime.utcnow(),
            duration_seconds=duration_seconds,
            reason=reason,
        )
        logger.debug(f"Cooldown set for {symbol}: {duration_seconds}s — {reason}")

    def is_in_cooldown(self, symbol: str) -> Tuple[bool, float]:
        entry = self.symbol_cooldowns.get(symbol)
        if entry is None:
            return False, 0.0
        elapsed = (datetime.utcnow() - entry.triggered_at).total_seconds()
        remaining = entry.duration_seconds - elapsed
        if remaining <= 0:
            del self.symbol_cooldowns[symbol]
            return False, 0.0
        return True, round(remaining, 1)

    def clear_cooldown(self, symbol: str):
        self.symbol_cooldowns.pop(symbol, None)

    def _reset_daily_counter_if_needed(self):
        now = datetime.utcnow()
        if now.date() > self.daily_trade_reset.date():
            self.daily_trade_count = 0
            self.daily_trade_reset = now

    def increment_trade_count(self):
        self._reset_daily_counter_if_needed()
        self.daily_trade_count += 1
        self.trade_timestamps.append(datetime.utcnow())

    def is_overtrading(self) -> Tuple[bool, str]:
        now = datetime.utcnow()

        minute_ago = now - timedelta(seconds=60)
        trades_last_minute = sum(
            1 for ts in self.trade_timestamps
            if ts > minute_ago
        )
        if trades_last_minute >= self.max_trades_per_minute:
            return True, f"Max trades/min reached: {trades_last_minute}/{self.max_trades_per_minute}"

        hour_ago = now - timedelta(seconds=self.overtrading_window_seconds)
        trades_last_hour = sum(
            1 for ts in self.trade_timestamps
            if ts > hour_ago
        )
        if trades_last_hour >= self.max_trades_per_hour:
            return True, f"Max trades/hour reached: {trades_last_hour}/{self.max_trades_per_hour}"

        self._reset_daily_counter_if_needed()
        if self.daily_trade_count >= self.max_daily_trades:
            return True, f"Max daily trades reached: {self.daily_trade_count}/{self.max_daily_trades}"

        return False, ""

    def is_duplicate_signal(self, symbol: str, action: str) -> bool:
        now = datetime.utcnow()
        key = f"{symbol}_{action}"
        last = self.recent_signals.get(key)
        if last is None:
            return False
        elapsed = (now - last["timestamp"]).total_seconds()
        if elapsed < self.signal_dedup_window_seconds:
            return True
        del self.recent_signals[key]
        return False

    def register_signal(self, symbol: str, action: str):
        key = f"{symbol}_{action}"
        self.recent_signals[key] = {
            "symbol": symbol,
            "action": action,
            "timestamp": datetime.utcnow(),
        }

    def has_conflicting_position(self, symbol: str, action: str, positions: list) -> bool:
        for p in positions:
            if p.get("symbol") != symbol:
                continue
            existing_type = p.get("order_type", "").upper()
            new_type = action.replace("OPEN_", "").upper()
            if existing_type and new_type and existing_type != new_type:
                return True
        return False

    def count_positions_for_symbol(self, symbol: str, positions: list) -> int:
        return sum(1 for p in positions if p.get("symbol") == symbol)

    async def is_execution_allowed(
        self,
        decision: dict,
        positions: list,
        account: dict,
    ) -> Tuple[bool, str]:
        symbol = decision.get("symbol", "")
        action = decision.get("action", "")
        self.check_kill_switch_auto_reset()

        if self.kill_switch_active:
            reason = f"Kill-switch active: {self.kill_switch_reason}"
            self._log_block(symbol, action, reason)
            return False, reason

        if symbol in self.blocked_symbols:
            reason = f"Symbol {symbol} is blocked"
            self._log_block(symbol, action, reason)
            return False, reason

        in_cd, remaining = self.is_in_cooldown(symbol)
        if in_cd:
            reason = f"Symbol {symbol} in cooldown ({remaining:.0f}s remaining)"
            self._log_block(symbol, action, reason)
            return False, reason

        over, over_reason = self.is_overtrading()
        if over:
            reason = f"Overtrading guard: {over_reason}"
            self._log_block(symbol, action, reason)
            if self.event_bus:
                await self.event_bus.publish(TradingEvent(
                    event_type=EventBus.OVERTRADING_ALERT,
                    source="ExecutionSafetyLayer",
                    payload={"reason": reason, "symbol": symbol},
                    priority=2,
                ))
            return False, reason

        if self.is_duplicate_signal(symbol, action):
            reason = f"Duplicate signal for {symbol}/{action} (dedup window)"
            self._log_block(symbol, action, reason)
            return False, reason

        if len(positions) >= self.max_concurrent_positions:
            reason = f"Max concurrent positions ({self.max_concurrent_positions}) reached"
            self._log_block(symbol, action, reason)
            return False, reason

        sym_count = self.count_positions_for_symbol(symbol, positions)
        if sym_count >= self.max_positions_per_symbol:
            reason = f"Max positions per symbol ({self.max_positions_per_symbol}) reached for {symbol}"
            self._log_block(symbol, action, reason)
            return False, reason

        if action.startswith("OPEN") and self.has_conflicting_position(symbol, action, positions):
            reason = f"Conflicting position exists for {symbol}"
            self._log_block(symbol, action, reason)
            return False, reason

        free_margin = account.get("free_margin", 0)
        balance = account.get("balance", 1)
        if free_margin < balance * 0.05:
            reason = f"Insufficient free margin: {free_margin:.2f} ({free_margin/balance:.1%} of balance)"
            self._log_block(symbol, action, reason)
            return False, reason

        confidence = decision.get("confidence", 0.0)
        if confidence < 0.55:
            reason = f"Confidence too low: {confidence:.2%} < 55%"
            self._log_block(symbol, action, reason)
            return False, reason

        self._log_allow(symbol, action, confidence)
        return True, "OK"

    def on_trade_executed(self, symbol: str, action: str, result: dict):
        self.increment_trade_count()
        self.register_signal(symbol, action)
        self.set_cooldown(symbol, self.default_cooldown_seconds, reason="post_execution")

    def on_trade_closed(self, symbol: str, profit: float):
        if profit < 0:
            self.set_cooldown(
                symbol,
                self.cooldown_after_loss_seconds,
                reason=f"post_loss profit={profit:.2f}",
            )
        else:
            self.set_cooldown(
                symbol,
                self.cooldown_after_win_seconds,
                reason=f"post_win profit={profit:.2f}",
            )

    def block_symbol(self, symbol: str, reason: str = "manual"):
        self.blocked_symbols.add(symbol)
        logger.warning(f"Symbol {symbol} blocked: {reason}")

    def unblock_symbol(self, symbol: str):
        self.blocked_symbols.discard(symbol)
        logger.info(f"Symbol {symbol} unblocked")

    def _log_block(self, symbol: str, action: str, reason: str):
        self.blocked_log.append({
            "symbol": symbol,
            "action": action,
            "reason": reason,
            "timestamp": datetime.utcnow().isoformat(),
        })

    def _log_allow(self, symbol: str, action: str, confidence: float):
        self.allowed_log.append({
            "symbol": symbol,
            "action": action,
            "confidence": confidence,
            "timestamp": datetime.utcnow().isoformat(),
        })

    def get_safety_report(self) -> dict:
        active_cooldowns = {}
        for symbol, entry in list(self.symbol_cooldowns.items()):
            in_cd, remaining = self.is_in_cooldown(symbol)
            if in_cd:
                active_cooldowns[symbol] = {
                    "remaining_seconds": remaining,
                    "reason": entry.reason,
                }

        now = datetime.utcnow()
        minute_ago = now - timedelta(seconds=60)
        hour_ago = now - timedelta(seconds=3600)

        return {
            "kill_switch_active": self.kill_switch_active,
            "kill_switch_reason": self.kill_switch_reason,
            "kill_switch_triggered_at": (
                self.kill_switch_triggered_at.isoformat()

    def _load_state_from_db(self):
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute(
                "SELECT ticket, symbol, order_type, volume, open_price, sl, tp, strategy, ai_reasoning "
                "FROM trades WHERE status='OPEN'"
            )
            rows = c.fetchall()
            for row in rows:
                ticket = row[0]
                self.open_trades[ticket] = {
                    "ticket": ticket,
                    "symbol": row[1],
                    "order_type": row[2],
                    "volume": row[3],
                    "open_price": row[4],
                    "sl": row[5],
                    "tp": row[6],
                    "strategy": row[7],
                    "ai_reasoning": row[8],
                    "open_time": datetime.utcnow().isoformat(),
                }
            c.execute(
                "SELECT symbol, order_type, profit, strategy FROM trades "
                "WHERE status='CLOSED' ORDER BY id DESC LIMIT 200"
            )
            closed = c.fetchall()
            for row in closed:
                self.memory_engine.record_trade({
                    "symbol": row[0],
                    "order_type": row[1],
                    "profit": row[2] or 0.0,
                    "strategy": row[3] or "UNKNOWN",
                })
                self.risk_engine.record_trade_result(row[2] or 0.0, 10000)
                self.strategy_engine.record_result(row[3] or "UNKNOWN", row[2] or 0.0)
            conn.close()
            logger.info(f"Loaded {len(self.open_trades)} open trades from DB")
        except Exception as e:
            logger.error(f"Load state from DB error: {e}")

    async def update_market_data(self, symbol: str, tick: dict, candles: dict):
        prev_regime = self.current_regime
        self.market_intelligence.update(symbol, tick, candles)
        clean_candles = {}
        for tf, c in candles.items():
            if c is not None and len(c) > 0:
                clean_candles[tf] = c
        if clean_candles:
            self._candles_cache[symbol] = clean_candles
        sd = self.market_intelligence.symbol_data.get(symbol, {})
        self._volatility_cache[symbol] = sd.get("volatility", 0.0)
        self._trend_cache[symbol] = sd.get("trend_strength", 0.0)

        try:
            if candles and any(v is not None for v in candles.values()):
                indicators = self.market_intelligence.get_technical_indicators(symbol, candles)
                regime, conf = self.regime_classifier.classify(symbol, candles, indicators)
                self._regime_confidence[symbol] = conf
                features = self.regime_classifier._build_classifier_features(candles, indicators)
                if features is not None:
                    self.regime_classifier.train_on_labeled(features, regime)

            new_regime = self.market_intelligence.symbol_data.get(symbol, {}).get("regime", "RANGING")
            if new_regime != prev_regime and new_regime != "UNKNOWN":
                self.current_regime = new_regime
                await self.event_bus.publish(TradingEvent(
                    event_type=EventBus.REGIME_CHANGED,
                    source="MasterAI",
                    payload={
                        "symbol": symbol,
                        "prev_regime": prev_regime,
                        "new_regime": new_regime,
                        "confidence": self._regime_confidence.get(symbol, 0.5),
                    },
                    priority=1,
                ))
                logger.info(f"Regime changed for {symbol}: {prev_regime} → {new_regime}")
        except Exception as e:
            logger.error(f"Update market data error ({symbol}): {e}")

    async def think_and_decide(
        self,
        positions: list,
        account: dict,
        market_data: dict,
    ) -> List[dict]:
        self.decision_cycles += 1
        raw_decisions = []

        try:
            if not account:
                return raw_decisions

            balance = account.get("balance", 10000)
            equity = account.get("equity", balance)
            self.risk_engine.update_equity(equity, balance)

            confidences = []

            for symbol in self.symbols:
                try:
                    candles = self._candles_cache.get(symbol)
                    if not candles:
                        continue

                    indicators = self.market_intelligence.get_technical_indicators(symbol, candles)
                    if not indicators:
                        continue

                    state = self.market_intelligence.build_state_vector(symbol, candles, account)
                    if state is None or len(state) == 0:
                        continue

                    market_summary = self.market_intelligence.get_market_summary(symbol)
                    regime = market_summary.get("regime", "RANGING")
                    volatility = market_summary.get("volatility", 0.0)
                    trend_strength = market_summary.get("trend_strength", 0.0)
                    momentum = market_summary.get("momentum", 0.0)

                    rsi_m15 = indicators.get("m15", {}).get("rsi", 50)

                    strategy = self.strategy_engine.select_strategy(
                        regime=regime,
                        volatility=volatility,
                        trend_strength=trend_strength,
                        momentum=momentum,
                        rsi=rsi_m15,
                        win_rate=self.risk_engine.win_rate,
                    )

                    trading_mode = self.strategy_engine.select_trading_mode(regime, volatility)
                    self.current_trading_mode = trading_mode
                    self._cycle_delay = self.strategy_engine.get_cycle_delay_for_mode(trading_mode)

                    signal = self.strategy_engine.generate_signal(
                        strategy=strategy,
                        indicators=indicators,
                        regime=regime,
                    )

                    rl_action, rl_quality = self.learning_engine.get_action_recommendation(
                        symbol, state
                    )

                    pattern_quality = self.memory_engine.get_pattern_quality(
                        symbol, strategy, regime
                    )

                    sentiment = self.self_development.get_market_sentiment(symbol)

                    decision = self.decision_engine.make_decision(
                        symbol=symbol,
                        state=state,
                        signal=signal,
                        rl_action=rl_action,
                        rl_quality=rl_quality,
                        market_data=market_data,
                        positions=positions,
                        account=account,
                        risk_engine=self.risk_engine,
                        strategy=strategy,
                        regime=regime,
                        indicators=indicators,
                        pattern_quality=pattern_quality,
                        sentiment=sentiment,
                    )

                    self._last_states[symbol] = state
                    self._last_actions[symbol] = rl_action

                    if decision:
                        raw_decisions.append(decision)

                    confidences.append(signal.get("confidence", 0.0))
                    self.current_strategy = strategy
                    self.current_regime = regime

                except Exception as e:
                    logger.error(f"Think and decide error for {symbol}: {e}")
                    continue

            if confidences:
                self.avg_confidence = float(np.mean(confidences))

            if self.decision_cycles % 20 == 0:
                logger.info(
                    f"AI cycle #{self.decision_cycles} | "
                    f"mode={self.current_trading_mode} | "
                    f"strategy={self.current_strategy} | "
                    f"regime={self.current_regime} | "
                    f"conf={self.avg_confidence:.2%} | "
                    f"decisions={len(raw_decisions)}"
                )

            if self.decision_cycles % 10 == 0:
                training_result = await self.learning_engine.train_agents(self.memory_engine)
                self.total_training_steps += 1

            for position in positions:
                symbol = position.get("symbol", "")
                ticket = position.get("ticket")
                if not ticket:
                    continue
                candles = self._candles_cache.get(symbol, {})
                m15_ind = self.market_intelligence.get_technical_indicators(
                    symbol, candles
                ).get("m15", {}) if candles else {}
                current_price = m15_ind.get("current_close", 0)
                if current_price <= 0:
                    md = market_data.get(symbol, {})
                    tick = md.get("tick", md)
                    current_price = tick.get("bid", 0)
                atr = m15_ind.get("atr", current_price * 0.001 if current_price > 0 else 0.001)
                if current_price > 0:
                    self.position_tracker.update_position(position, current_price, atr)

            balance_val = account.get("balance", 0)
            equity_val = account.get("equity", balance_val)
            if balance_val > 0:
                self.analytics_engine.record_equity_point(equity_val, balance_val, datetime.utcnow())

            safe_decisions = []
            for decision in raw_decisions:
                try:
                    symbol = decision.get("symbol", "")
                    action = decision.get("action", "")

                    allowed, reason = await self.execution_safety.is_execution_allowed(
                        decision=decision,
                        positions=positions,
                        account=account,
                    )
                    if not allowed:
                        logger.debug(f"Decision blocked by safety layer: {symbol}/{action} — {reason}")
                        continue

                    candles = self._candles_cache.get(symbol, {})
                    m15_ind = self.market_intelligence.get_technical_indicators(
                        symbol, candles
                    ).get("m15", {}) if candles else {}
                    entry_price = m15_ind.get("current_close", decision.get("entry_price", 0))
                    sl_price = decision.get("sl", 0.0)
                    base_volume = decision.get("volume", 0.01)

                    if entry_price > 0:
                        adj_volume, size_reason = self.advanced_risk_engine.suggest_position_size(
                            symbol=symbol,
                            account=account,
                            positions=positions,
                            entry_price=entry_price,
                            sl_price=sl_price,
                            base_volume=base_volume,
                        )
                        decision["volume"] = adj_volume
                        decision["volume_adjusted_reason"] = size_reason

                    state = self._last_states.get(symbol)
                    if state is not None:
                        regime = decision.get("market_regime", self.current_regime)
                        strategy_val = decision.get("strategy", self.current_strategy)
                        win_prob, ml_confidence = self.online_learning_engine.predict_trade_quality(
                            features=state,
                            regime=regime,
                            strategy=strategy_val,
                        )
                        decision["ml_win_probability"] = round(win_prob, 3)
                        decision["ml_confidence"] = round(ml_confidence, 3)

                        if win_prob < 0.35 and ml_confidence > 0.6:
                            logger.debug(
                                f"Decision blocked by ML: {symbol} win_prob={win_prob:.2f} "
                                f"ml_conf={ml_confidence:.2f}"
                            )
                            continue

                    await self.event_bus.publish(TradingEvent(
                        event_type=EventBus.SIGNAL_GENERATED,
                        source="MasterAI",
                        payload={
                            "symbol": symbol,
                            "action": action,
                            "confidence": decision.get("confidence"),
                            "strategy": decision.get("strategy"),
                            "regime": decision.get("market_regime"),
                            "volume": decision.get("volume"),
                        },
                        priority=1,
                    ))

                    safe_decisions.append(decision)

                except Exception as e:
                    logger.error(f"Final think decision filter error ({decision.get('symbol')}): {e}")
                    continue

            return safe_decisions

        except Exception as e:
            logger.error(f"MasterAI think_and_decide critical error: {e}")
            return []

    async def manage_open_positions(
        self,
        positions: list,
        account: dict,
        market_data: dict,
        modify_fn: Callable,
        close_fn: Callable,
    ):
        try:
            for position in positions:
                try:
                    symbol = position.get("symbol")
                    if not symbol:
                        continue

                    candles = self._candles_cache.get(symbol)
                    if not candles:
                        continue

                    indicators = self.market_intelligence.get_technical_indicators(symbol, candles)
                    market_summary = self.market_intelligence.get_market_summary(symbol)
                    regime = market_summary.get("regime", "RANGING")

                    m15 = indicators.get("m15", {})
                    current_price = m15.get("current_close", 0)
                    if current_price <= 0:
                        md = market_data.get(symbol, {})
                        tick = md.get("tick", md)
                        current_price = tick.get("bid", 0) or tick.get("ask", 0)
                    if current_price <= 0:
                        continue

                    mgmt_decision = self.decision_engine.make_position_management_decision(
                        position=position,
                        current_price=current_price,
                        indicators=indicators,
                        regime=regime,
                        risk_engine=self.risk_engine,
                        account=account,
                    )

                    if mgmt_decision is None:
                        continue

                    action = mgmt_decision.get("action")
                    ticket = mgmt_decision.get("ticket")

                    if action == "MODIFY_SLTP" and ticket:
                        result = modify_fn(
                            ticket=ticket,
                            sl=mgmt_decision.get("new_sl", 0.0),
                            tp=mgmt_decision.get("new_tp", 0.0),
                        )
                        if result.get("success"):
                            logger.info(
                                f"Position {ticket} modified: "
                                f"SL={mgmt_decision.get('new_sl')} "
                                f"TP={mgmt_decision.get('new_tp')} | "
                                f"{mgmt_decision.get('reasoning', '')}"
                            )
                            if self.broadcast_callback:
                                await self.broadcast_callback({
                                    "type": "position_modified",
                                    "ticket": ticket,
                                    "new_sl": mgmt_decision.get("new_sl"),
                                    "new_tp": mgmt_decision.get("new_tp"),
                                    "reasoning": mgmt_decision.get("reasoning"),
                                    "timestamp": datetime.utcnow().isoformat(),
                                })

                    elif action == "CLOSE" and ticket:
                        result = close_fn(ticket=ticket)
                        if result.get("success"):
                            profit = result.get("profit", 0.0)
                            logger.info(
                                f"Position {ticket} closed by AI | "
                                f"profit={profit:.2f} | "
                                f"{mgmt_decision.get('reasoning', '')}"
                            )
                            await self.on_trade_closed(ticket, profit)
                            if self.broadcast_callback:
                                await self.broadcast_callback({
                                    "type": "trade_closed",
                                    "ticket": ticket,
                                    "profit": profit,
                                    "reasoning": mgmt_decision.get("reasoning"),
                                    "timestamp": datetime.utcnow().isoformat(),
                                })

                except Exception as e:
                    logger.error(f"Manage position error for {position.get('ticket')}: {e}")
                    continue

        except Exception as e:
            logger.error(f"Manage open positions error: {e}")

    async def on_trade_opened(self, ticket: int, trade: dict):
        try:
            self.open_trades[ticket] = trade
            symbol = trade.get("symbol", "")
            state = self._last_states.get(symbol)
            action = self._last_actions.get(symbol, 0)
            if state is not None:
                zero_next = np.zeros_like(state)
                self.memory_engine.store_experience(
                    state=state,
                    action=action,
                    reward=0.0,
                    next_state=zero_next,
                    done=False,
                    metadata={"symbol": symbol, "ticket": ticket},
                )

            self.execution_safety.on_trade_executed(symbol, trade.get("order_type", "BUY"), {"ticket": ticket})

            await self.event_bus.publish(TradingEvent(
                event_type=EventBus.TRADE_OPENED,
                source="MasterAI",
                payload={
                    "ticket": ticket,
                    "symbol": symbol,
                    "action": trade.get("order_type"),
                    "volume": trade.get("volume"),
                    "open_price": trade.get("open_price"),
                    "strategy": trade.get("strategy"),
                },
                priority=2,
            ))

            if state is not None:
                record = FeatureRecord(
                    timestamp=datetime.utcnow().isoformat(),
                    symbol=symbol,
                    regime=self.current_regime,
                    strategy=self.current_strategy,
                    features=state.tolist(),
                    label=0,
                    confidence=trade.get("confidence", 0.5),
                    profit=0.0,
                )
                self.online_learning_engine.store_feature(record)

            logger.info(f"Trade opened: ticket={ticket} symbol={symbol}")
        except Exception as e:
            logger.error(f"on_trade_opened error: {e}")

    async def on_trade_closed(self, ticket: int, profit: float):
        try:
            trade = self.open_trades.pop(ticket, None)
            if trade is None:
                trade = {"ticket": ticket, "profit": profit}

            symbol = trade.get("symbol", "")
            strategy = trade.get("strategy", "UNKNOWN")

            balance = 10000
            self.risk_engine.record_trade_result(profit, balance)
            self.strategy_engine.record_result(strategy, profit)

            open_time_str = trade.get("open_time", datetime.utcnow().isoformat())
            try:
                open_time = datetime.fromisoformat(open_time_str)
                duration_hours = (datetime.utcnow() - open_time).total_seconds() / 3600
            except Exception:
                duration_hours = 1.0

            reward = self.memory_engine.calculate_reward({
                "profit": profit,
                "balance": balance,
                "duration_hours": duration_hours,
            })

            state = self._last_states.get(symbol, np.zeros(50))
            action = self._last_actions.get(symbol, 0)
            next_state = np.zeros_like(state)

            self.memory_engine.store_experience(
                state=state,
                action=action,
                reward=reward,
                next_state=next_state,
                done=True,
                metadata={
                    "symbol": symbol,
                    "ticket": ticket,
                    "profit": profit,
                    "strategy": strategy,
                },
            )

            self.memory_engine.record_trade({
                "ticket": ticket,
                "symbol": symbol,
                "profit": profit,
                "strategy": strategy,
                "regime": self.current_regime,
            })

            await self.learning_engine.train_agents(self.memory_engine)

            self.execution_safety.on_trade_closed(symbol, profit)

            if state is not None:
                label = 1 if profit > 0 else 0
                record = FeatureRecord(
                    timestamp=datetime.utcnow().isoformat(),
                    symbol=symbol,
                    regime=self.current_regime,
                    strategy=self.current_strategy,
                    features=state.tolist(),
                    label=label,
                    confidence=0.5,
                    profit=profit,
                )
                self.online_learning_engine.online_update(record)

            final_metrics = self.position_tracker.remove_position(ticket)
            if final_metrics:
                trade_record = {
                    "ticket": ticket,
                    "profit": profit,
                    "symbol": final_metrics.get("symbol", "UNKNOWN"),
                    "strategy": self.current_strategy,
                    "regime": self.current_regime,
                    "open_time": final_metrics.get("open_time", datetime.utcnow().isoformat()),
                    "close_time": datetime.utcnow().isoformat(),
                    "peak_profit": final_metrics.get("peak_profit", 0.0),
                    "mae": final_metrics.get("max_adverse_excursion", 0.0),
                    "mfe": final_metrics.get("max_favorable_excursion", 0.0),
                    "age_hours": final_metrics.get("age_hours", 0.0),
                }
                self.analytics_engine.record_trade_result(trade_record)

                candles = self._candles_cache.get(symbol, {})
                if candles:
                    indicators = self.market_intelligence.get_technical_indicators(symbol, candles)
                    features = self.regime_classifier._build_classifier_features(candles, indicators)
                    if features is not None:
                        regime_to_label = self.current_regime if profit > 0 else "VOLATILE"
                        self.regime_classifier.train_on_labeled(features, regime_to_label)

            await self.event_bus.publish(TradingEvent(
                event_type=EventBus.TRADE_CLOSED,
                source="MasterAI",
                payload={
                    "ticket": ticket,
                    "symbol": symbol,
                    "profit": profit,
                    "strategy": self.current_strategy,
                    "regime": self.current_regime,
                },
                priority=2,
            ))

            logger.info(
                f"Trade closed: ticket={ticket} profit={profit:.2f} "
                f"reward={reward:.3f} strategy={strategy}"
            )
        except Exception as e:
            logger.error(f"on_trade_closed error: {e}")

    async def self_development_loop(self):
        logger.info("Self-development loop started")
        while True:
            try:
                await self.self_development.run_development_cycle(
                    market_intelligence=self.market_intelligence,
                    strategy_engine=self.strategy_engine,
                    memory_engine=self.memory_engine,
                )

                for suggestion in self.self_development.improvement_suggestions:
                    stype = suggestion.get("type")
                    if stype == "RISK_REDUCTION":
                        if self.risk_engine.risk_per_trade > 0.005:
                            self.risk_engine.risk_per_trade *= 0.95
                            logger.info(
                                f"Self-development: risk reduced to "
                                f"{self.risk_engine.risk_per_trade:.3%}"
                            )
                    elif stype == "RISK_INCREASE":
                        if self.risk_engine.risk_per_trade < 0.02:
                            self.risk_engine.risk_per_trade *= 1.02
                            logger.info(
                                f"Self-development: risk increased to "
                                f"{self.risk_engine.risk_per_trade:.3%}"
                            )

    async def _risk_monitoring_loop(self):
        logger.info("Risk monitoring loop started")
        while True:
            try:
                await asyncio.sleep(30)
                from backend import mt5_integration
                positions = mt5_integration.get_positions() if mt5_integration.connected else []
                account = mt5_integration.get_account_info() if mt5_integration.connected else {}

                if account and positions is not None:
                    for symbol in self.symbols:
                        candles = self._candles_cache.get(symbol, {})
                        m15 = candles.get("m15")
                        if m15 is not None and len(m15) >= 2:
                            try:
                                closes = [c[4] for c in m15[-5:]]
                                if len(closes) >= 2 and closes[-2] > 0:
                                    ret = (closes[-1] - closes[-2]) / closes[-2]
                                    self.advanced_risk_engine.record_symbol_return(symbol, ret)
                            except Exception:
                                pass

                    await self.advanced_risk_engine.emit_risk_alerts(
                        self.event_bus, positions, account
                    )

                    balance = account.get("balance", 0)
                    equity = account.get("equity", balance)
                    if balance > 0:
                        dd = (balance - equity) / balance
                        if dd > 0.20 and not self.execution_safety.kill_switch_active:
                            await self.execution_safety.activate_kill_switch(
                                f"Emergency stop: drawdown={dd:.2%} > 20%"
                            )

            except asyncio.CancelledError:
                break
            except ImportError:
                await asyncio.sleep(60)
            except Exception as e:
                logger.error(f"Risk monitoring loop error: {e}")
                await asyncio.sleep(30)

class ReconciliationEngine:
    """
    Production-grade reconciliation engine.
    Synchronizes internal state with MT5 broker.
    Critical for ensuring system-broker consistency.
    """
    
    def __init__(self, db_path: str, event_bus: Optional["EventBus"] = None):
        self.db_path = db_path
        self.event_bus = event_bus
        self._lock = asyncio.Lock()
        self.reconciliation_history: deque = deque(maxlen=100)
        self.last_reconciliation: Optional[datetime] = None
        self.consistency_status: ConsistencyStatus = ConsistencyStatus.CONSISTENT
        
        self._init_reconciliation_tables()
        logger.info("ReconciliationEngine initialized")
    
    def _init_reconciliation_tables(self):
        """Create reconciliation tracking tables."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                CREATE TABLE IF NOT EXISTS reconciliation_history (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp TEXT,
                    total_broker INTEGER,
                    total_internal INTEGER,
                    matched INTEGER,
                    orphan_broker TEXT,
                    orphan_internal TEXT,
                    repairs TEXT,
                    consistent INTEGER,
                    duration_ms REAL
                )
            """)
            c.execute("""
                CREATE TABLE IF NOT EXISTS position_snapshot (
                    ticket INTEGER PRIMARY KEY,
                    symbol TEXT,
                    volume REAL,
                    open_price REAL,
                    sl REAL,
                    tp REAL,
                    snapshot_time TEXT,
                    source TEXT
                )
            """)
            c.execute("""
                CREATE TABLE IF NOT EXISTS reconciliation_repair_log (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp TEXT,
                    ticket INTEGER,
                    action TEXT,
                    reason TEXT,
                    success INTEGER
                )
            """)
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Init reconciliation tables error: {e}")
    
    async def take_broker_snapshot(self, get_broker_positions_fn: Callable) -> Dict[int, dict]:
        """Take a snapshot of current broker positions."""
        try:
            positions = await get_broker_positions_fn() if asyncio.iscoroutinefunction(get_broker_positions_fn) else get_broker_positions_fn()
            snapshot = {}
            for pos in positions:
                ticket = pos.get("ticket")
                if ticket:
                    snapshot[ticket] = {
                        "ticket": ticket,
                        "symbol": pos.get("symbol"),
                        "volume": pos.get("volume", 0.0),
                        "open_price": pos.get("open_price", 0.0),
                        "sl": pos.get("sl", 0.0),
                        "tp": pos.get("tp", 0.0),
                        "order_type": pos.get("order_type"),
                        "profit": pos.get("profit", 0.0),
                    }
            
            # Save snapshot to DB
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            for ticket, data in snapshot.items():
                c.execute("""
                    INSERT OR REPLACE INTO position_snapshot (ticket, symbol, volume, open_price, sl, tp, snapshot_time, source)
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """, (ticket, data.get("symbol"), data.get("volume"), data.get("open_price"),
                      data.get("sl"), data.get("tp"), datetime.utcnow().isoformat(), "broker"))
            conn.commit()
            conn.close()
            
            return snapshot
        except Exception as e:
            logger.error(f"Take broker snapshot error: {e}")
            return {}
    
    async def take_internal_snapshot(self, get_internal_positions_fn: Callable) -> Dict[int, dict]:
        """Take a snapshot of current internal positions from DB."""
        try:
            positions = await get_internal_positions_fn() if asyncio.iscoroutinefunction(get_internal_positions_fn) else get_internal_positions_fn()
            snapshot = {}
            for pos in positions:
                ticket = pos.get("ticket")
                if ticket:
                    snapshot[ticket] = {
                        "ticket": ticket,
                        "symbol": pos.get("symbol"),
                        "volume": pos.get("volume", 0.0),
                        "open_price": pos.get("open_price", 0.0),
                        "sl": pos.get("sl", 0.0),
                        "tp": pos.get("tp", 0.0),
                        "order_type": pos.get("order_type"),
                        "profit": pos.get("profit", 0.0),
                    }
            
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            for ticket, data in snapshot.items():
                c.execute("""
                    INSERT OR REPLACE INTO position_snapshot (ticket, symbol, volume, open_price, sl, tp, snapshot_time, source)
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """, (ticket, data.get("symbol"), data.get("volume"), data.get("open_price"),
                      data.get("sl"), data.get("tp"), datetime.utcnow().isoformat(), "internal"))
            conn.commit()
            conn.close()
            
            return snapshot
        except Exception as e:
            logger.error(f"Take internal snapshot error: {e}")
            return {}
    
    async def reconcile(
        self,
        broker_positions: List[dict],
        internal_positions: List[dict],
        auto_repair: bool = True,
    ) -> ReconciliationReport:
        """
        Perform full reconciliation between broker and internal state.
        
        Args:
            broker_positions: Positions from MT5
            internal_positions: Positions from internal DB
            auto_repair: If True, attempt automatic repair of discrepancies
        
        Returns:
            ReconciliationReport with full details
        """
        start_time = time.time()
        
        async with self._lock:
            broker_map = {p["ticket"]: p for p in broker_positions if p.get("ticket")}
            internal_map = {p["ticket"]: p for p in internal_positions if p.get("ticket")}
            
            broker_tickets = set(broker_map.keys())
            internal_tickets = set(internal_map.keys())
            
            orphan_broker = list(broker_tickets - internal_tickets)
            orphan_internal = list(internal_tickets - broker_tickets)
            common_tickets = broker_tickets & internal_tickets
            
            mismatches = []
            repairs_performed = []
            
            # Check common positions for mismatches
            for ticket in common_tickets:
                broker = broker_map[ticket]
                internal = internal_map[ticket]
                mismatch = self._compare_positions(ticket, broker, internal)
                if mismatch.status != ConsistencyStatus.CONSISTENT:
                    mismatches.append(mismatch)
                    if auto_repair:
                        repair_result = await self._attempt_repair(mismatch, broker, internal)
                        if repair_result:
                            repairs_performed.append(repair_result)
                            mismatch.resolved = True
                            mismatch.resolution_action = repair_result
            
            # Determine overall consistency
            consistent = len(orphan_broker) == 0 and len(orphan_internal) == 0 and len(mismatches) == 0
            requires_manual = len(orphan_broker) > 0 or len([m for m in mismatches if not m.resolved]) > 0
            
            if consistent:
                self.consistency_status = ConsistencyStatus.CONSISTENT
            elif orphan_broker:
                self.consistency_status = ConsistencyStatus.BROKER_HAS_EXTRA
            elif orphan_internal:
                self.consistency_status = ConsistencyStatus.INTERNAL_HAS_EXTRA
            elif mismatches:
                self.consistency_status = ConsistencyStatus.VOLUME_MISMATCH
            
            report = ReconciliationReport(
                timestamp=datetime.utcnow(),
                total_broker_positions=len(broker_tickets),
                total_internal_positions=len(internal_tickets),
                matched_positions=len(common_tickets),
                orphan_broker_positions=orphan_broker,
                orphan_internal_positions=orphan_internal,
                mismatches=mismatches,
                repairs_performed=repairs_performed,
                consistency_achieved=consistent,
                requires_manual_intervention=requires_manual,
                duration_ms=(time.time() - start_time) * 1000,
            )
            
            # Store history
            self.reconciliation_history.append(report)
            self.last_reconciliation = report.timestamp
            
            # Save to DB
            self._save_reconciliation_report(report)
            
            # Emit event if inconsistency detected
            if not consistent and self.event_bus:
                await self.event_bus.publish(TradingEvent(
                    event_type="reconciliation_issue",
                    source="ReconciliationEngine",
                    payload={
                        "orphan_broker": orphan_broker,
                        "orphan_internal": orphan_internal,
                        "mismatches": len(mismatches),
                        "requires_manual": requires_manual,
                    },
                    priority=2,
                ))
            
            logger.info(
                f"Reconciliation completed: broker={len(broker_tickets)}, "
                f"internal={len(internal_tickets)}, matched={len(common_tickets)}, "
                f"orphan_broker={len(orphan_broker)}, orphan_internal={len(orphan_internal)}, "
                f"mismatches={len(mismatches)}, consistent={consistent}, duration={report.duration_ms:.0f}ms"
            )
            
            return report
    
    def _compare_positions(self, ticket: int, broker: dict, internal: dict) -> PositionReconciliationResult:
        """Compare broker and internal position details."""
        status = ConsistencyStatus.CONSISTENT
        
        # Check volume mismatch (allow 0.01 tolerance)
        if abs(broker.get("volume", 0) - internal.get("volume", 0)) > 0.009:
            status = ConsistencyStatus.VOLUME_MISMATCH
        
        # Check price mismatch (allow 0.5 pip tolerance)
        price_tolerance = 0.00005 if "JPY" not in broker.get("symbol", "") else 0.05
        if abs(broker.get("open_price", 0) - internal.get("open_price", 0)) > price_tolerance:
            status = ConsistencyStatus.PRICE_MISMATCH
        
        return PositionReconciliationResult(
            ticket=ticket,
            symbol=broker.get("symbol", ""),
            broker_exists=True,
            internal_exists=True,
            broker_volume=broker.get("volume", 0),
            internal_volume=internal.get("volume", 0),
            broker_open_price=broker.get("open_price", 0),
            internal_open_price=internal.get("open_price", 0),
            broker_sl=broker.get("sl", 0),
            internal_sl=internal.get("sl", 0),
            broker_tp=broker.get("tp", 0),
            internal_tp=internal.get("tp", 0),
            status=status,
        )
    
    async def _attempt_repair(self, mismatch: PositionReconciliationResult, broker: dict, internal: dict) -> Optional[str]:
        """Attempt to automatically repair a mismatch."""
        try:
            if mismatch.status == ConsistencyStatus.VOLUME_MISMATCH:
                # Volume mismatch - internal should be updated to match broker
                conn = sqlite3.connect(self.db_path)
                c = conn.cursor()
                c.execute(
                    "UPDATE trades SET volume = ? WHERE ticket = ?",
                    (broker.get("volume", 0), mismatch.ticket)
                )
                conn.commit()
                conn.close()
                
                self._log_repair(mismatch.ticket, "UPDATE_VOLUME", 
                                f"Volume corrected from {mismatch.internal_volume} to {mismatch.broker_volume}", True)
                return f"VOLUME_CORRECTED:{mismatch.ticket}"
            
            elif mismatch.status == ConsistencyStatus.PRICE_MISMATCH:
                # Price mismatch - log but don't auto-repair (broker is source of truth)
                self._log_repair(mismatch.ticket, "PRICE_MISMATCH_LOG", 
                                f"Price diff: broker={mismatch.broker_open_price}, internal={mismatch.internal_open_price}", True)
                return f"PRICE_MISMATCH_LOGGED:{mismatch.ticket}"
            
            return None
        except Exception as e:
            logger.error(f"Repair attempt failed for ticket {mismatch.ticket}: {e}")
            self._log_repair(mismatch.ticket, "REPAIR_FAILED", str(e), False)
            return None
    
    def _log_repair(self, ticket: int, action: str, reason: str, success: bool):
        """Log repair action to database."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO reconciliation_repair_log (timestamp, ticket, action, reason, success)
                VALUES (?, ?, ?, ?, ?)
            """, (datetime.utcnow().isoformat(), ticket, action, reason, 1 if success else 0))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Log repair error: {e}")
    
    def _save_reconciliation_report(self, report: ReconciliationReport):
        """Save reconciliation report to database."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO reconciliation_history 
                (timestamp, total_broker, total_internal, matched, orphan_broker, orphan_internal, repairs, consistent, duration_ms)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                report.timestamp.isoformat(),
                report.total_broker_positions,
                report.total_internal_positions,
                report.matched_positions,
                json.dumps(report.orphan_broker_positions),
                json.dumps(report.orphan_internal_positions),
                json.dumps(report.repairs_performed),
                1 if report.consistency_achieved else 0,
                report.duration_ms,
            ))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Save reconciliation report error: {e}")
    
    async def validate_consistency(self) -> Tuple[bool, str]:
        """
        Quick consistency validation before enabling trading.
        Returns (is_consistent, reason).
        """
        if not self.last_reconciliation:
            return False, "No reconciliation performed yet"
        
        if self.consistency_status != ConsistencyStatus.CONSISTENT:
            return False, f"System inconsistent: {self.consistency_status.value}"
        
        # Check if last reconciliation is too old (> 5 minutes)
        if self.last_reconciliation and (datetime.utcnow() - self.last_reconciliation).total_seconds() > 300:
            return False, "Last reconciliation is stale (> 5 minutes)"
        
        return True, "System consistent"
    
    def get_last_report(self) -> Optional[ReconciliationReport]:
        """Get the most recent reconciliation report."""
        if self.reconciliation_history:
            return self.reconciliation_history[-1]
        return None

# 2. EVENT DURABILITY LAYER
# ============================================================================

class EventDeliveryStatus(Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"
    DEAD_LETTER = "dead_letter"


@dataclass
class DurableEvent:
    """Persistent event with delivery tracking."""
    event_id: str
    event_type: str
    source: str
    payload: Dict[str, Any]
    created_at: datetime
    status: EventDeliveryStatus
    retry_count: int = 0
    last_attempt: Optional[datetime] = None
    last_error: Optional[str] = None
    next_retry_at: Optional[datetime] = None


class EventDurabilityLayer:
    """
    Production-grade durable event queue.
    Provides at-least-once delivery with retries and dead-letter queue.
    """
    
    def __init__(self, db_path: str, event_bus: Optional["EventBus"] = None):
        self.db_path = db_path
        self.event_bus = event_bus
        self._processing = False
        self._processor_task: Optional[asyncio.Task] = None
        
        # Retry configuration
        self.base_retry_delay_seconds: float = 1.0
        self.max_retry_delay_seconds: float = 300.0
        self.max_retry_count: int = 5
        self.batch_size: int = 50
        self.processing_interval_seconds: float = 1.0
        
        self._init_durability_tables()
        logger.info("EventDurabilityLayer initialized")
    
    def _init_durability_tables(self):
        """Create durable event tables."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                CREATE TABLE IF NOT EXISTS durable_events (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    event_id TEXT UNIQUE,
                    event_type TEXT,
                    source TEXT,
                    payload TEXT,
                    created_at TEXT,
                    status TEXT,
                    retry_count INTEGER DEFAULT 0,
                    last_attempt TEXT,
                    last_error TEXT,
                    next_retry_at TEXT
                )
            """)
            c.execute("""
                CREATE TABLE IF NOT EXISTS dead_letter_queue (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    event_id TEXT,
                    event_type TEXT,
                    source TEXT,
                    payload TEXT,
                    failed_at TEXT,
                    failure_reason TEXT,
                    retry_count INTEGER
                )
            """)
            c.execute("""
                CREATE INDEX IF NOT EXISTS idx_durable_events_status ON durable_events(status, next_retry_at)
            """)
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Init durability tables error: {e}")
    
    async def publish_durable(self, event: TradingEvent) -> str:
        """
        Publish an event with durability guarantee.
        Returns event_id.
        """
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO durable_events 
                (event_id, event_type, source, payload, created_at, status, retry_count)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (
                event.event_id,
                event.event_type,
                event.source,
                json.dumps(event.payload, default=str),
                event.timestamp,
                EventDeliveryStatus.PENDING.value,
                0,
            ))
            conn.commit()
            conn.close()
            
            logger.debug(f"Durable event published: {event.event_id} ({event.event_type})")
            return event.event_id
            
        except sqlite3.IntegrityError:
            logger.warning(f"Event {event.event_id} already exists, skipping")
            return event.event_id
        except Exception as e:
            logger.error(f"Publish durable event error: {e}")
            raise
    
    async def start_processor(self, handler_fn: Callable[[TradingEvent], Awaitable[bool]]):
        """
        Start the background processor that consumes durable events.
        
        Args:
            handler_fn: Async function that takes TradingEvent and returns bool (success/failure)
        """
        self._processing = True
        self._processor_task = asyncio.create_task(self._process_loop(handler_fn))
        logger.info("Event durability processor started")
    
    async def stop_processor(self):
        """Stop the background processor."""
        self._processing = False
        if self._processor_task and not self._processor_task.done():
            self._processor_task.cancel()
            try:
                await self._processor_task
            except asyncio.CancelledError:
                pass
        logger.info("Event durability processor stopped")
    
    async def _process_loop(self, handler_fn: Callable[[TradingEvent], Awaitable[bool]]):
        """Main processing loop for durable events."""
        while self._processing:
            try:
                # Get pending events ready for retry
                events = await self._get_pending_events()
                
                for event_data in events:
                    if not self._processing:
                        break
                    
                    event = self._deserialize_event(event_data)
                    
                    # Mark as processing
                    await self._update_event_status(event.event_id, EventDeliveryStatus.PROCESSING)
                    
                    try:
                        success = await handler_fn(event)
                        
                        if success:
                            await self._mark_completed(event.event_id)
                            logger.debug(f"Durable event processed: {event.event_id}")
                        else:
                            await self._handle_failure(event, "Handler returned False")
                            
                    except Exception as e:
                        logger.error(f"Error processing durable event {event.event_id}: {e}")
                        await self._handle_failure(event, str(e))
                
                await asyncio.sleep(self.processing_interval_seconds)
                
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"Event durability processor loop error: {e}")
                await asyncio.sleep(5.0)
    
    async def _get_pending_events(self) -> List[dict]:
        """Get pending events that are ready for processing."""
        try:
            conn = sqlite3.connect(self.db_path)
            conn.row_factory = sqlite3.Row
            c = conn.cursor()
            
            now = datetime.utcnow().isoformat()
            c.execute("""
                SELECT * FROM durable_events 
                WHERE status = ? 
                AND (next_retry_at IS NULL OR next_retry_at <= ?)
                ORDER BY created_at ASC
                LIMIT ?
            """, (EventDeliveryStatus.PENDING.value, now, self.batch_size))
            
            rows = c.fetchall()
            conn.close()
            return [dict(row) for row in rows]
            
        except Exception as e:
            logger.error(f"Get pending events error: {e}")
            return []
    
    async def _update_event_status(self, event_id: str, status: EventDeliveryStatus):
        """Update event status."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                UPDATE durable_events 
                SET status = ?, last_attempt = ?
                WHERE event_id = ?
            """, (status.value, datetime.utcnow().isoformat(), event_id))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Update event status error: {e}")
    
    async def _mark_completed(self, event_id: str):
        """Mark event as completed and remove from queue."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                DELETE FROM durable_events WHERE event_id = ?
            """, (event_id,))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Mark completed error: {e}")
    
    async def _handle_failure(self, event: DurableEvent, error: str):
        """Handle event processing failure with retry logic."""
        new_retry_count = event.retry_count + 1
        
        if new_retry_count >= self.max_retry_count:
            # Move to dead-letter queue
            await self._move_to_dlq(event, error)
            await self._delete_event(event.event_id)
            logger.warning(f"Event {event.event_id} moved to DLQ after {new_retry_count} retries")
        else:
            # Calculate next retry time with exponential backoff
            delay = min(
                self.base_retry_delay_seconds * (2 ** (new_retry_count - 1)),
                self.max_retry_delay_seconds
            )
            next_retry_at = datetime.utcnow() + timedelta(seconds=delay)
            
            try:
                conn = sqlite3.connect(self.db_path)
                c = conn.cursor()
                c.execute("""
                    UPDATE durable_events 
                    SET status = ?, retry_count = ?, last_error = ?, next_retry_at = ?
                    WHERE event_id = ?
                """, (
                    EventDeliveryStatus.PENDING.value,
                    new_retry_count,
                    error[:500],
                    next_retry_at.isoformat(),
                    event.event_id,
                ))
                conn.commit()
                conn.close()
                logger.debug(f"Event {event.event_id} scheduled for retry in {delay}s (attempt {new_retry_count})")
            except Exception as e:
                logger.error(f"Handle failure update error: {e}")
    
    async def _move_to_dlq(self, event: DurableEvent, error: str):
        """Move failed event to dead-letter queue."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO dead_letter_queue 
                (event_id, event_type, source, payload, failed_at, failure_reason, retry_count)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (
                event.event_id,
                event.event_type,
                event.source,
                json.dumps(event.payload, default=str),
                datetime.utcnow().isoformat(),
                error[:500],
                event.retry_count,
            ))
            conn.commit()
            conn.close()
            
            if self.event_bus:
                await self.event_bus.publish(TradingEvent(
                    event_type="dead_letter_event",
                    source="EventDurabilityLayer",
                    payload={
                        "event_id": event.event_id,
                        "event_type": event.event_type,
                        "failure_reason": error,
                    },
                    priority=2,
                ))
        except Exception as e:
            logger.error(f"Move to DLQ error: {e}")
    
    async def _delete_event(self, event_id: str):
        """Delete event from durable queue."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("DELETE FROM durable_events WHERE event_id = ?", (event_id,))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Delete event error: {e}")
    
    def _deserialize_event(self, event_data: dict) -> DurableEvent:
        """Deserialize database row to DurableEvent."""
        return DurableEvent(
            event_id=event_data["event_id"],
            event_type=event_data["event_type"],
            source=event_data["source"],
            payload=json.loads(event_data["payload"]),
            created_at=datetime.fromisoformat(event_data["created_at"]),
            status=EventDeliveryStatus(event_data["status"]),
            retry_count=event_data["retry_count"],
            last_attempt=datetime.fromisoformat(event_data["last_attempt"]) if event_data.get("last_attempt") else None,
            last_error=event_data.get("last_error"),
            next_retry_at=datetime.fromisoformat(event_data["next_retry_at"]) if event_data.get("next_retry_at") else None,
        )
    
    async def replay_dead_letter(self, event_id: Optional[str] = None) -> int:
        """
        Replay events from dead-letter queue.
        Returns number of events replayed.
        """
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            
            if event_id:
                c.execute("SELECT * FROM dead_letter_queue WHERE event_id = ?", (event_id,))
            else:
                c.execute("SELECT * FROM dead_letter_queue ORDER BY failed_at ASC")
            
            rows = c.fetchall()
            conn.close()
            
            replayed = 0
            for row in rows:
                # Re-insert into durable queue
                conn2 = sqlite3.connect(self.db_path)
                c2 = conn2.cursor()
                try:
                    c2.execute("""
                        INSERT INTO durable_events 
                        (event_id, event_type, source, payload, created_at, status, retry_count)
                        VALUES (?, ?, ?, ?, ?, ?, ?)
                    """, (
                        row[1], row[2], row[3], row[4],
                        datetime.utcnow().isoformat(),
                        EventDeliveryStatus.PENDING.value,
                        0,
                    ))
                    c2.execute("DELETE FROM dead_letter_queue WHERE event_id = ?", (row[1],))
                    conn2.commit()
                    replayed += 1
                except sqlite3.IntegrityError:
                    logger.warning(f"Event {row[1]} already in queue, skipping")
                finally:
                    conn2.close()
            
            logger.info(f"Replayed {replayed} events from DLQ")
            return replayed
            
        except Exception as e:
            logger.error(f"Replay dead letter error: {e}")
            return 0
    
    async def get_stats(self) -> dict:
        """Get queue statistics."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            
            c.execute("SELECT COUNT(*) FROM durable_events WHERE status = ?", (EventDeliveryStatus.PENDING.value,))
            pending = c.fetchone()[0]
            
            c.execute("SELECT COUNT(*) FROM durable_events WHERE status = ?", (EventDeliveryStatus.PROCESSING.value,))
            processing = c.fetchone()[0]
            
            c.execute("SELECT COUNT(*) FROM dead_letter_queue")
            dead_letter = c.fetchone()[0]
            
            conn.close()
            
            return {
                "pending_events": pending,
                "processing_events": processing,
                "dead_letter_count": dead_letter,
                "max_retry_count": self.max_retry_count,
                "processor_running": self._processing,
            }
        except Exception as e:
            logger.error(f"Get stats error: {e}")
            return {"error": str(e)}

@dataclass
class IdempotencyRecord:
    """Record of an idempotent operation."""
    idempotency_key: str
    operation_type: str
    symbol: str
    parameters: Dict[str, Any]
    result: Dict[str, Any]
    created_at: datetime
    executed_at: datetime
    success: bool


class IdempotencyLayer:
    """
    Idempotency layer for trade executions.
    Prevents duplicate orders after crashes or network issues.
    """
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self._cache: Dict[str, IdempotencyRecord] = {}
        self._cache_ttl_seconds: int = 3600  # 1 hour cache
        self._cleanup_interval_seconds: int = 300  # 5 minutes
        
        self._init_idempotency_tables()
        self._start_cleanup_task()
        logger.info("IdempotencyLayer initialized")
    
    def _init_idempotency_tables(self):
        """Create idempotency tables."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                CREATE TABLE IF NOT EXISTS idempotency_records (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    idempotency_key TEXT UNIQUE,
                    operation_type TEXT,
                    symbol TEXT,
                    parameters TEXT,
                    result TEXT,
                    created_at TEXT,
                    executed_at TEXT,
                    success INTEGER
                )
            """)
            c.execute("""
                CREATE INDEX IF NOT EXISTS idx_idempotency_key ON idempotency_records(idempotency_key)
            """)
            c.execute("""
                CREATE INDEX IF NOT EXISTS idx_idempotency_created ON idempotency_records(created_at)
            """)
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"Init idempotency tables error: {e}")
    
    def generate_key(self, operation_type: str, symbol: str, params: Dict[str, Any]) -> str:
        """
        Generate an idempotency key for an operation.
        Deterministic based on operation characteristics.
        """
        # Normalize params (sort keys, convert to string)
        normalized_params = json.dumps(params, sort_keys=True, default=str)
        
        # Create unique string
        unique_string = f"{operation_type}:{symbol}:{normalized_params}"
        
        # Add timestamp window (minute-level granularity to prevent duplicates within same minute)
        # This allows natural retries within the same minute to be deduplicated
        timestamp_window = datetime.utcnow().strftime("%Y%m%d%H%M")
        unique_string_with_time = f"{unique_string}:{timestamp_window}"
        
        # Generate SHA256 hash
        return hashlib.sha256(unique_string_with_time.encode()).hexdigest()[:32]
    
    async def check_and_store(
        self,
        idempotency_key: str,
        operation_type: str,
        symbol: str,
        parameters: Dict[str, Any],
        executor: Callable[[], Awaitable[Dict[str, Any]]],
    ) -> Tuple[Dict[str, Any], bool]:
        """
        Execute operation with idempotency guarantee.
        
        Returns:
            (result, was_cached) - result and whether it came from cache
        """
        # Check cache first
        cached = await self._get_cached_result(idempotency_key)
        if cached:
            logger.info(f"Idempotency cache hit for key {idempotency_key[:8]}...")
            return cached, True
        
        # Check database for existing execution
        existing = await self._get_existing_record(idempotency_key)
        if existing:
            logger.info(f"Idempotency DB hit for key {idempotency_key[:8]}...")
            await self._cache_record(existing)
            return existing.result, True
        
        # Execute fresh
        try:
            result = await executor()
            success = result.get("success", False)
            
            record = IdempotencyRecord(
                idempotency_key=idempotency_key,
                operation_type=operation_type,
                symbol=symbol,
                parameters=parameters,
                result=result,
                created_at=datetime.utcnow(),
                executed_at=datetime.utcnow(),
                success=success,
            )
            
            await self._save_record(record)
            await self._cache_record(record)
            
            logger.info(f"Idempotency fresh execution for key {idempotency_key[:8]}: success={success}")
            return result, False
            
        except Exception as e:
            logger.error(f"Idempotency execution error: {e}")
            raise
    
    async def _get_cached_result(self, key: str) -> Optional[Dict[str, Any]]:
        """Get result from in-memory cache."""
        record = self._cache.get(key)
        if record:
            # Check TTL
            age = (datetime.utcnow() - record.executed_at).total_seconds()
            if age < self._cache_ttl_seconds:
                return record.result
            else:
                # Expired, remove from cache
                del self._cache[key]
        return None
    
    async def _get_existing_record(self, key: str) -> Optional[IdempotencyRecord]:
        """Get existing record from database."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute(
                "SELECT operation_type, symbol, parameters, result, created_at, executed_at, success "
                "FROM idempotency_records WHERE idempotency_key = ?",
                (key,)
            )
            row = c.fetchone()
            conn.close()
            
            if row:
                return IdempotencyRecord(
                    idempotency_key=key,
                    operation_type=row[0],
                    symbol=row[1],
                    parameters=json.loads(row[2]),
                    result=json.loads(row[3]),
                    created_at=datetime.fromisoformat(row[4]),
                    executed_at=datetime.fromisoformat(row[5]),
                    success=bool(row[6]),
                )
            return None
        except Exception as e:
            logger.error(f"Get existing record error: {e}")
            return None
    
    async def _save_record(self, record: IdempotencyRecord):
        """Save record to database."""
        try:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("""
                INSERT INTO idempotency_records 
                (idempotency_key, operation_type, symbol, parameters, result, created_at, executed_at, success)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                record.idempotency_key,
                record.operation_type,
                record.symbol,
                json.dumps(record.parameters, default=str),
                json.dumps(record.result, default=str),
                record.created_at.isoformat(),
                record.executed_at.isoformat(),
                1 if record.success else 0,
            ))
            conn.commit()
            conn.close()
        except sqlite3.IntegrityError:
            logger.warning(f"Record {record.idempotency_key} already exists, skipping")
        except Exception as e:
            logger.error(f"Save record error: {e}")
    
    async def _cache_record(self, record: IdempotencyRecord):
        """Cache record in memory."""
        self._cache[record.idempotency_key] = record
        
        # Trim cache if too large
        if len(self._cache) > 10000:
            # Remove oldest entries
            oldest_keys = sorted(self._cache.keys(), 
                                key=lambda k: self._cache[k].executed_at)[:1000]
            for key in oldest_keys:
                del self._cache[key]
    
    def _start_cleanup_task(self):
        """Start background cleanup task for old records."""
        asyncio.create_task(self._cleanup_loop())
    
    async def _cleanup_loop(self):
        """Background cleanup of old idempotency records."""
        while True:
            try:
                await asyncio.sleep(self._cleanup_interval_seconds)
                
                # Delete records older than 7 days
                cutoff = (datetime.utcnow() - timedelta(days=7)).isoformat()
                
                conn = sqlite3.connect(self.db_path)
                c = conn.cursor()
                c.execute("DELETE FROM idempotency_records WHERE created_at < ?", (cutoff,))
                deleted = c.rowcount
                conn.commit()
                conn.close()
                
                if deleted > 0:
                    logger.debug(f"Cleaned up {deleted} old idempotency records")
                    
            except asyncio.CancelledError:
                break
            except Exception as e:
                logger.error(f"Idempotency cleanup error: {e}")
    
    async def get_stats(self) -> dict:
        """Get idempotency layer statistics."""
        return {
            "cache_size": len(self._cache),
            "cache_ttl_seconds": self._cache_ttl_seconds,
        }

# 4. STARTUP RECOVERY SEQUENCE
# ============================================================================

class StartupRecoverySequence:
    """
    Orchestrates the complete startup recovery process.
    Ensures system consistency before trading is enabled.
    """
    
    def __init__(
        self,
        persistence_layer: PersistenceLayer,
        reconciliation_engine: ReconciliationEngine,
        event_durability: EventDurabilityLayer,
        idempotency_layer: IdempotencyLayer,
        execution_safety: ExecutionSafetyLayer,
        event_bus: EventBus,
    ):
        self.persistence = persistence_layer
        self.reconciliation = reconciliation_engine
        self.event_durability = event_durability
        self.idempotency = idempotency_layer
        self.execution_safety = execution_safety
        self.event_bus = event_bus
        
        self.recovery_status: Dict[str, Any] = {}
        self.trading_enabled: bool = False
        self.recovery_complete: bool = False
    
    async def execute_recovery(
        self,
        master_ai: "MasterAI",
        get_broker_positions_fn: Callable,
        get_internal_positions_fn: Callable,
    ) -> Tuple[bool, str]:
        """
        Execute full startup recovery sequence.
        
        Returns:
            (success, message) - whether recovery succeeded and trading can be enabled
        """
        logger.info("=" * 60)
        logger.info("STARTING PRODUCTION RECOVERY SEQUENCE")
        logger.info("=" * 60)
        
        start_time = datetime.utcnow()
        self.recovery_status = {
            "started_at": start_time.isoformat(),
            "steps": {},
        }
        
        # Step 1: Load persisted state
        step1_ok = await self._step1_load_state(master_ai)
        if not step1_ok:
            return False, "Failed to load persisted state"
        
        # Step 2: Reconcile with broker
        step2_ok, step2_msg = await self._step2_reconcile_broker(
            get_broker_positions_fn, get_internal_positions_fn
        )
        if not step2_ok:
            return False, f"Reconciliation failed: {step2_msg}"
        
        # Step 3: Replay critical events
        step3_ok = await self._step3_replay_events()
        if not step3_ok:
            return False, "Failed to replay events"
        
        # Step 4: Validate system consistency
        step4_ok, step4_msg = await self._step4_validate_consistency()
        if not step4_ok:
            return False, f"Consistency validation failed: {step4_msg}"
        
        # Step 5: Enable trading
        await self._step5_enable_trading()
        
        # Record completion
        self.recovery_complete = True
        duration = (datetime.utcnow() - start_time).total_seconds()
        self.recovery_status["completed_at"] = datetime.utcnow().isoformat()
        self.recovery_status["duration_seconds"] = duration
        self.recovery_status["trading_enabled"] = self.trading_enabled
        
        logger.info("=" * 60)
        logger.info(f"RECOVERY COMPLETE: duration={duration:.2f}s, trading_enabled={self.trading_enabled}")
        logger.info("=" * 60)
        
        # Emit recovery complete event
        await self.event_bus.publish(TradingEvent(
            event_type="recovery_complete",
            source="StartupRecoverySequence",
            payload={
                "duration_seconds": duration,
                "trading_enabled": self.trading_enabled,
                "status": self.recovery_status,
            },
            priority=1,
        ))
        
        return self.trading_enabled, "Recovery successful, trading enabled" if self.trading_enabled else "Recovery completed but trading not enabled"
    
    async def _step1_load_state(self, master_ai: "MasterAI") -> bool:
        """Step 1: Load persisted state from database."""
        logger.info("STEP 1: Loading persisted state...")
        
        try:
            load_results = await self.persistence.load_all(master_ai)
            self.recovery_status["steps"]["load_state"] = {
                "status": "success",
                "results": {k: v for k, v in load_results.items()},
            }
            logger.info(f"Step 1 complete: loaded {len(load_results)} components")
            return True
        except Exception as e:
            logger.error(f"Step 1 failed: {e}")
            self.recovery_status["steps"]["load_state"] = {
                "status": "failed",
                "error": str(e),
            }
            return False
    
    async def _step2_reconcile_broker(
        self,
        get_broker_positions_fn: Callable,
        get_internal_positions_fn: Callable,
    ) -> Tuple[bool, str]:
        """Step 2: Reconcile with broker positions."""
        logger.info("STEP 2: Reconciling with broker...")
        
        try:
            # Get positions
            if asyncio.iscoroutinefunction(get_broker_positions_fn):
                broker_positions = await get_broker_positions_fn()
            else:
                broker_positions = get_broker_positions_fn()
            
            if asyncio.iscoroutinefunction(get_internal_positions_fn):
                internal_positions = await get_internal_positions_fn()
            else:
                internal_positions = get_internal_positions_fn()
            
            # Perform reconciliation
            report = await self.reconciliation.reconcile(
                broker_positions=broker_positions,
                internal_positions=internal_positions,
                auto_repair=True,
            )
            
            self.recovery_status["steps"]["reconcile"] = {
                "status": "success" if report.consistency_achieved else "warning",
                "broker_positions": report.total_broker_positions,
                "internal_positions": report.total_internal_positions,
                "matched": report.matched_positions,
                "orphan_broker": len(report.orphan_broker_positions),
                "orphan_internal": len(report.orphan_internal_positions),
                "mismatches": len(report.mismatches),
                "repairs": len(report.repairs_performed),
                "requires_manual": report.requires_manual_intervention,
            }
            
            if report.requires_manual_intervention:
                logger.warning(f"Step 2 complete with manual intervention required: {len(report.orphan_broker_positions)} orphans")
                return False, f"Manual intervention required: orphan positions exist"
            
            if not report.consistency_achieved:
                logger.warning(f"Step 2 complete with inconsistencies: {len(report.mismatches)} mismatches")
                return False, f"Inconsistencies detected: {len(report.mismatches)} mismatches"
            
            logger.info("Step 2 complete: system consistent with broker")
            return True, "OK"
            
        except Exception as e:
            logger.error(f"Step 2 failed: {e}")
            self.recovery_status["steps"]["reconcile"] = {
                "status": "failed",
                "error": str(e),
            }
            return False, str(e)
    
    async def _step3_replay_events(self) -> bool:
        """Step 3: Replay critical events from dead-letter queue."""
        logger.info("STEP 3: Replaying critical events...")
        
        try:
            replayed = await self.event_durability.replay_dead_letter()
            self.recovery_status["steps"]["replay_events"] = {
                "status": "success",
                "replayed_count": replayed,
            }
            logger.info(f"Step 3 complete: replayed {replayed} events")
            return True
        except Exception as e:
            logger.error(f"Step 3 failed: {e}")
            self.recovery_status["steps"]["replay_events"] = {
                "status": "failed",
                "error": str(e),
            }
            return False
    
    async def _step4_validate_consistency(self) -> Tuple[bool, str]:
        """Step 4: Validate overall system consistency."""
        logger.info("STEP 4: Validating system consistency...")
        
        try:
            # Check reconciliation consistency
            rec_consistent, rec_reason = await self.reconciliation.validate_consistency()
            if not rec_consistent:
                return False, f"Reconciliation: {rec_reason}"
            
            # Check execution safety status
            if self.execution_safety.kill_switch_active:
                return False, f"Kill switch active: {self.execution_safety.kill_switch_reason}"
            
            # Check for blocked symbols
            if self.execution_safety.blocked_symbols:
                logger.warning(f"Blocked symbols exist: {self.execution_safety.blocked_symbols}")
            
            self.recovery_status["steps"]["validate"] = {
                "status": "success",
                "checks": {
                    "reconciliation": rec_consistent,
                    "kill_switch": not self.execution_safety.kill_switch_active,
                },
            }
            
            logger.info("Step 4 complete: all consistency checks passed")
            return True, "OK"
            
        except Exception as e:
            logger.error(f"Step 4 failed: {e}")
            self.recovery_status["steps"]["validate"] = {
                "status": "failed",
                "error": str(e),
            }
            return False, str(e)
    
    async def _step5_enable_trading(self):
        """Step 5: Enable trading (kill-switch deactivation)."""
        logger.info("STEP 5: Enabling trading...")
        
        try:
            # Deactivate kill switch if active
            if self.execution_safety.kill_switch_active:
                self.execution_safety.deactivate_kill_switch(manual=False)
                logger.info("Kill switch deactivated")
            
            self.trading_enabled = True
            
            self.recovery_status["steps"]["enable_trading"] = {
                "status": "success",
                "trading_enabled": True,
            }
            
            logger.info("Step 5 complete: trading enabled")
            
        except Exception as e:
            logger.error(f"Step 5 failed: {e}")
            self.recovery_status["steps"]["enable_trading"] = {
                "status": "failed",
                "error": str(e),
            }
            self.trading_enabled = False
    
    def get_recovery_status(self) -> dict:
        """Get current recovery status."""
        return {
            "recovery_complete": self.recovery_complete,
            "trading_enabled": self.trading_enabled,
            "recovery_status": self.recovery_status,
        }

    self,

    volume: float,
    sl: float = 0.0,
    tp: float = 0.0,
    comment: str = "MasterAI",
    strategy: str = "",
) -> dict:

        "symbol": symbol,

    }

        executor=execute,
    )
    
    if was_cached:

    return result

# Patch 3: Add idempotent close position method
async def _idempotent_close_position(self, ticket: int, volume: Optional[float] = None) -> dict:
    """
    Close position with idempotency guarantee.
    """
    params = {"ticket": ticket, "volume": volume}
    
    # Find symbol for this ticket
    symbol = "UNKNOWN"
    for sym, pos in self.position_tracker.tracked_positions.items():
        if pos.get("ticket") == ticket:
            symbol = pos.get("symbol", "UNKNOWN")
            break
    
    idempotency_key = self.idempotency_layer.generate_key(
        operation_type="CLOSE_POSITION",
        symbol=symbol,
        params=params,
    )
    
    async def execute():
        from backend import mt5_integration
        return mt5_integration.close_position(ticket=ticket, volume=volume)
    
    result, was_cached = await self.idempotency_layer.check_and_store(
        idempotency_key=idempotency_key,
        operation_type="CLOSE_POSITION",
        symbol=symbol,
        parameters=params,
        executor=execute,
    )
    
    if was_cached:
        logger.info(f"Close position deduplicated for ticket {ticket}")
    
    return result

MasterAI.idempotent_close_position = _idempotent_close_position
